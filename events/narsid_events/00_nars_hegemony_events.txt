namespace = nars_hegemony

# Day 0: Setup
nars_hegemony.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		add_character_flag = nikator_aspirant

		every_vassal_or_below = {
			limit = {
				is_ai = no
			}
            trigger_event = {
                id = nars_hegemony.0002
                days = 3
            }
        }

		every_in_list = {
			list = selian_coalition_members
			limit = {
				is_ai = no
			}
            trigger_event = {
                id = nars_hegemony.0003
                days = 3
            }
		}

		trigger_event = {
            id = nars_hegemony.0004
            days = 7
        }
	}
}

# Day 3: Vassalized and Tributaries Notice Event
nars_hegemony.0002 = {
    type = character_event
    title = nars_hegemony.0002.t
    desc = nars_hegemony.0002.desc
    theme = nars_war

    left_portrait = {
        character = root
        animation = celebrate_sword
    }

    lower_right_portrait = {
        character = scope:nikator_aspirant
    }

    immediate = {
        every_independent_ruler = {
            limit = {
                has_character_modifier = nars_nikator_aspirant_modifier
            }
            save_scope_as = nikator_aspirant
        }
    }

    option = { # Submit and help the Hegemon-Aspirant!
        name = nars_hegemony.0002.a
        custom_tooltip = nars_hegemony_reject_tt
        scope:nikator_aspirant = {
            add_to_variable_list = { name = selian_coalition_members target = root }
        }
        ai_chance = {
            base = 100
        }
    }
}

	type = character_event
	title = nars_hegemony.0003.t
	desc = nars_hegemony.0003.desc
	theme = nars_war

	left_portrait = {
		character = root
		animation = celebrate_sword
	}

	lower_right_portrait = {
		character = scope:nikator_aspirant
		animation = celebrate_sword
	}

	immediate = {
		every_independent_ruler = {
			limit = {
				has_character_modifier = nars_nikator_aspirant_modifier
			}
			save_scope_as = nikator_aspirant
		}
	}

	option = { # Fight them
		name = nars_hegemony.0003.b
		flavor = nars_hegemony.0003.b.flavor
		custom_tooltip = nars_hegemony_reject_tt     

		scope:nikator_aspirant = {
			add_to_variable_list = { name = selian_coalition_members target = root }
		}

		if = {
			limit = {
				any_vassal = { vassal_stance = glory_hound }
			}
			every_vassal = {
				limit = { vassal_stance = glory_hound }
				custom = every_glory_hound_vassal
				add_opinion = {
					modifier = approving_opinion
					opinion = 15
					target = root
					months = 46
				}
			}
		}  
	}
}

# Day 7: War Declaration
nars_hegemony.0004 = {
	hidden = yes
	type = character_event

	immediate = {
		every_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					is_independent_ruler = yes
					NOT = { has_alliance = root }
					NOT = { has_truce = root }
				}
				add_to_variable_list = selian_coalition_members
		}
		if = {
			limit = {
				any_in_list = { list = selian_coalition_members }
			}
			
			ordered_in_list = {
				list = selian_coalition_members
				order_by = current_military_strength
				save_scope_as = coalition_war_leader
			}

			scope:coalition_war_leader = {
				start_war = {
					target = root
					target_title = root.primary_title
					cb = selian_defiance_cb
				}
			}
			root = {
				random_character_war = {
					limit = { primary_attacker = scope:coalition_war_leader }
					every_in_list = {
						list = selian_coalition_members
						limit = {
							NOT = { this = scope:coalition_war_leader }
						}
						prev = {
							add_attacker = prev
						}
					}
				}
			}
		}
	}
}
