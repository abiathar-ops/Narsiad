nars_faith.1041 = {
	type = character_event
	title = nars_faith.1041.t
	desc = nars_faith.1041.desc
	theme = administrative
	override_background = fp3_zoroastrian_temple
	override_effect_2d = { reference = smoke }
	widget = { gui = event_window_widget_vfx_fog container = foreground_shader_vfx_container }
	left_portrait = {
		character = root
		animation = prayer
	}
	right_portrait = {
		character = scope:evangelizer
		animation = wedding_priest
	}
	lower_right_portrait = scope:pagan

	immediate = {
		if = {
			limit = {
				is_narsocratic_e_h = yes
			}
			random_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					valid_for_minoan_conversion_trigger = yes
				}
				save_scope_as = pagan
			}
		}
		else = {
			random_neighboring_and_across_water_realm_same_rank_owner = {
				limit = {
					valid_for_minoan_conversion_trigger = yes
				}
				save_scope_as = pagan
				create_character = {
					location = root.location
					template = minoan_template.1011
					culture = root.culture
					faith = root.faith
					gender_female_chance = 100
					save_scope_as = evangelizer
					after_creation = {
						add_trait = nars_naturalis
					}
				}				
			}
		}
		scope:pagan.primary_title = { save_scope_as = title_to_convert }
		random_pool_character = {
			province = root.location
			limit = {
				faith = root.faith
				learning >= decent_skill_rating
			}
			save_scope_as = evangelizer
		}
		if = {
			limit = {
				NOT = { exists = scope:evangelizer }
			}
			create_character = {
				location = root.location
				culture = root.culture
				faith = root.faith
				template = minoan_template.1011
				gender_female_chance = 100
				save_scope_as = evangelizer
				after_creation = {
					add_trait = nars_naturalis
				}
			}
		}
		add_courtier = scope:evangelizer
		save_scope_as = sender
		location = { save_scope_as = starting_point }
	}

	option = {
		name = nars_faith.1041.a
		custom_tooltip = nars_faith.1041.a.tt
		scope:evangelizer = {
			start_travel_plan = {
				destination = scope:pagan.capital_province
				on_arrival_event = nars_faith.1042
				on_arrival_destinations = all
				return_trip = no # One way
			}
		}
	}
}

				}
			}
		}
	}

nars_faith.1043 = {
	type = character_event
	title = nars_faith.1041.t
	desc = nars_faith.1043.desc
	theme = administrative
	
	trigger = {
		scope:evangelizer = { is_alive = yes }
	}

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:evangelizer
		animation = wedding_priest
	}
	lower_right_portrait = scope:sender

	#Option A: hear them out
	option = {
		name = nars_faith.1043.a
		create_character_memory = {
			type = was_evangelized_memory
			participants = { missionary = scope:evangelizer sender = scope:sender }
		}
		set_character_faith_with_conversion = scope:evangelizer.faith
		if = {
			limit = { 
				scope:evangelizer.faith = { has_doctrine = doctrine_monotheist }
			}
			custom_description_no_bullet = { text = mandala_monotheist_warning_tt }
		}
		scope:evangelizer = {
			add_character_flag = success_conversion
		}
		scope:sender = {
			create_character_memory = {
				type = evangelized_memory
				participants = { missionary = scope:evangelizer pagan = scope:pagan }
			}
		}
		stress_impact = {
			zealous = major_stress_impact_gain
			content = major_stress_impact_gain
			craven = medium_stress_impact_gain
			arrogant = minor_stress_impact_gain
		}
	}

	#Option C: no thanks with stab
	option = {
		name = nars_faith.1043.c
		flavor = nars_faith.1043.c.tt
		add_piety = major_piety_gain
		scope:evangelizer = {
			death = {
				death_reason = death_execution
				killer = root
			}
		}
		scope:sender = {
			trigger_event = {
				id = nars_faith.1047
				days = 3
			}
		}
		scope:pagan = {
			add_character_flag = do_not_disturb_preachers
		}
		scope:title_to_convert = {
			set_variable = do_not_disturb_preachers
		}
		stress_impact = {
			compassionate = major_stress_impact_gain
			calm = major_stress_impact_gain
			gregarious = medium_stress_impact_gain
			zealous = medium_stress_impact_gain
		}
	}

	#Option B: no thanks
	option = {
		name = nars_faith.1043.b
		add_piety = medium_piety_gain
		scope:evangelizer = {
			add_character_flag = failed_conversion
		}
	}

	after = {
		if = {
			limit = {
				scope:sender = { is_alive = yes }
				scope:evangelizer = { is_alive = yes }
			}
			scope:evangelizer = {
				start_travel_plan = {
					destination = scope:sender.location
					on_arrival_event = nars_faith.1045
					on_arrival_destinations = all
					return_trip = no # One way
				}
			}
		}
		else_if = {
			limit = {
				scope:evangelizer = { is_alive = yes }
			}
			scope:evangelizer = {
				start_travel_plan = {
					destination = scope:starting_point
					on_arrival_event = nars_faith.1045
					on_arrival_destinations = all
					return_trip = no # One way
				}
			}
		}
	}
}
