# Sponsor Missionaries
nars_faith.0001 = {
    type = character_event
    title = nars_faith.0001.t
    desc = nars_faith.0001.desc
    theme = nars_faith
    
    left_portrait = {
        character = root
        animation = prayer
    }
    right_portrait = {
        character = scope:evangelizer
        animation = wedding_priest
    }

    lower_right_portrait = scope:pagan

    trigger = {
        nars_faith_scheme_has_targets_trigger = yes
    }

    immediate = {
        random_sub_realm_title = {
            limit = {
                tier = tier_county
                NOT = { faith = root.faith } 
                holder = {
                    valid_for_selian_conversion_trigger = yes
                }
            }
            save_scope_as = title_to_convert
            holder = {
                save_scope_as = pagan
            }
        }
        if = {
            limit = {
                scope:pagan = { nars_faith_needs_bribe_trigger = yes }
            }
            save_scope_value_as = { name = needs_bribe value = yes }
        }
        else = {
            save_scope_value_as = { name = needs_bribe value = no }
        }
		hidden_effect = {
			random_pool_character = {
				province = root.location
				limit = {
					faith = faith:nars_religion
					learning >= decent_skill_rating
				}
				save_scope_as = evangelizer
			}

			if = {
				limit = {
					NOT = { exists = scope:evangelizer }
				}
				create_character = {
					location = root.location
					culture = culture:selian
					faith = faith:nars_religion
					template = selian_template.1011
					gender_female_chance = 100
					save_scope_as = evangelizer
					after_creation = {
						add_trait = nars_naturalis
					}
				}
			}
			add_courtier = scope:evangelizer
			save_scope_as = sender
			location = { save_scope_as = starting_point }
		}
    }

    option = {
        name = nars_faith.0001.a
		flavor = nars_faith.0001.a.flavor
        custom_tooltip = nars_faith.0001.a.tt
        pay_short_term_gold = {
            target = scope:evangelizer
            gold = major_gold_value
        }
        add_piety = minor_piety_gain
        if = {
            limit = {
                any_vassal = { has_vassal_stance = zealot }
            }
            every_vassal = {
                limit = { has_vassal_stance = zealot }
                custom = every_zealot_vassal
                add_opinion = {
                    modifier = approved_opinion
                    target = root
                    opinion = 5
                }
            }
        }
        scope:evangelizer = {
            start_travel_plan = {
                destination = scope:pagan.capital_province
                on_arrival_event = nars_faith.0002
                on_arrival_destinations = all
                return_trip = no # One way
            }
        }
    }
}

nars_faith.0002 = {
    hidden = yes

    immediate = {
        scope:pagan = { add_visiting_courtier = scope:evangelizer }
        if = {
            limit = {
                scope:pagan = {
                    OR = {
                        is_alive = no
                        is_landed = no
                    }
                }
            }
            scope:title_to_convert.holder = { save_scope_as = pagan }
        }

        if = {
            limit = {
                NOT = {
                    scope:title_to_convert.holder = {
                        valid_for_selian_conversion_trigger = yes
                    }
                }
            }
            scope:evangelizer = {
                start_travel_plan = {
                    destination = scope:starting_point
                    return_trip = no
                }
            }
        }
        else = {
            scope:title_to_convert.faith = { save_scope_as = old_faith }
            # Player event for scope:pagan
            if = {
                limit = {
                    scope:pagan = { is_ai = no }
                }
                scope:pagan = {
                    trigger_event = {
                        id = nars_faith.0003
                        days = 1
                    }
                }
            }
            # AI event for the evangelizer
            else = {
                scope:evangelizer = {
                    trigger_event = {
                        id = nars_faith.0004
                        days = 1
                    }
                }
            }
        }
    }
}

        name = nars_faith.0003.b
        flavor = nars_faith.0003.b.flavor

        scope:title_to_convert = {
            set_county_faith = scope:evangelizer.faith
        }

        scope:evangelizer = {
            add_character_flag = success_conversion
        }

        if = {
            limit = { exists = liege }
            liege = {
                add_opinion = {

                    modifier = approval_opinion
                    opinion = 15
                    years = 3
                }
            } 
        }
        
        stress_impact = {
            zealous = medium_stress_impact_gain
            content = medium_stress_impact_gain
            arrogant = minor_stress_impact_gain
            greedy = medium_stress_impact_gain
        }
    }

        name = nars_faith.0003.c
        flavor = nars_faith.0003.c.flavor

            add_character_flag = failed_conversion

            }

        if = {
            limit = { exists = liege }
            liege = {
                add_opinion = {
                    target = root
                    modifier = refused_bribe_opinion
                    opinion = -30
                    years = 3

            }

    }

    after = {
        if = {
            limit = {
                scope:sender = { is_alive = yes }

                    on_arrival_destinations = all

                }
            }
        }
        else_if = {
            limit = {
                scope:evangelizer = { is_alive = yes }
            }
            scope:evangelizer = {
                start_travel_plan = {
                    destination = scope:starting_point
                    on_arrival_event = nars_faith.0005
                    on_arrival_destinations = all
                    return_trip = no # One way
                }
            }
        }
    }
