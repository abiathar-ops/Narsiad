namespace = nars_artifacts

nars_artifacts.1001 = {
	type = character_event

	title = nars_artifacts.1001.t
	desc = nars_artifacts.1001.desc
	theme = emperor
	override_background = armory
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = personality_dishonorable
	}

	immediate = {
		play_music_cue = mx_cue_touching_moment
	}

	option = {
		# Sacrifice courtiers — brutal but FAST (30 days)
		name = nars_artifacts.1001.a
		add_tyranny = 15
		stress_impact = {
			just = minor_stress_impact_gain
			compassionate = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		random_courtier_or_guest = {
			limit = {
				is_adult = yes
				is_ethnically_arcadian = yes
				NOT = { is_close_family_of = root }
				NOT = { is_spouse_of = root }
			}
			death = {
				death_reason = death_murder
				killer = root
			}
		}
		hidden_effect = {
			# Lock the decision immediately — the forge is committed
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mend_narsil_decision
			}
			set_global_variable = {
				name = flag_mend_narsil_decision
				value = root
			}
			trigger_event = {
				id = nars_artifacts.1002
				days = 30
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}

	option = {
		name = nars_artifacts.1001.b
		remove_treasury_or_gold = 750
		stress_impact = {
			sadistic = minor_stress_impact_gain
			callous = miniscule_stress_impact_gain
			greedy = miniscule_stress_impact_gain
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mend_narsil_decision
			}
			set_global_variable = {
				name = flag_mend_narsil_decision
				value = root
			}
			trigger_event = {
				id = nars_artifacts.1003
				days = 120
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_trait = greedy
			}
		}
	}

	option = {
		# Not yet — decision remains available
		name = nars_artifacts.1001.c
		add_stress = miniscule_stress_impact_gain
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}
}
