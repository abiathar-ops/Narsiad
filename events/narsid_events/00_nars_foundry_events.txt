nars_narsil.10 = {
	type = character_event
	window = big_event_window
	title = nars_narsil.10.t
	desc = nars_narsil.10.desc
	theme = nars_war

	center_portrait = {
		character = root
		animation = personality_zealous
		camera = camera_body_right
	}

	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}

	trigger = {
	}

	immediate = {
		scope:owner = {
			save_scope_as = owner
		}
	}

	option = {
		name = nars_narsil.10.a
		flavor = nars_narsil_intro_tt
		add_character_modifier = {
			modifier = nars_resolute_restoration_personal_modifier
			years = 5
		}
		scope:scheme = {
			add_scheme_modifier = {
				type = nars_resolute_restoration_modifier
				years = 5
			}
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = nars_narsil.10.b
		flavor = nars_narsil_intro_tt
		custom_tooltip = no_change_restoration_tt
		ai_chance = {
			base = 100
		}
	}

	after = {
		add_character_flag = narsil_scheme_intensity_chosen
	}
}

nars_narsil.9000 = {
	type = character_event
	title = nars_narsil.9000.t
	desc = nars_narsil.9000.desc
	
	window = big_event_window

	theme = nars_war
	widget = { gui = event_window_widget_vfx_godray container = foreground_shader_vfx_container }

	center_portrait = {
		character = root
		camera = camera_scheme_widget_microscopic
		animation = marshal
	}

	immediate = {
		play_music_cue = mx_cue_positive_effect
		every_character_artifact = {
			limit = { has_variable = can_be_used_for_narsil } 
			destroy_artifact = this
		}
	}
	
	option = {
		name = nars_narsil.9000.a
		flavor = nars_narsil.9000.flavor
		create_narsil_effect = yes
		hidden_effect = {
			scope:scheme = { end_scheme = yes }
			scope:narsil_story = { end_story = yes }
			add_to_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:flag_mend_narsil_decision
			}
		}	
	}
}
