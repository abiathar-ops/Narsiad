			base = 0

nars_artifacts.1011 = {
	hidden = yes

	immediate = {
		scope:scheme = {
			scheme_owner = {
				save_scope_as = owner
				random = {
					chance = scope:scheme.scheme_success_chance
					save_scope_value_as = {
						name = scheme_successful
						value = yes
					}
				}
				if = {
					limit = { exists = scope:scheme_successful }
					trigger_event = {
						id = nars_artifacts.1021
					}
				}
				else = {
					trigger_event = {
						id = nars_artifacts.1022
					}	
				}
			}
		}
	}
}
