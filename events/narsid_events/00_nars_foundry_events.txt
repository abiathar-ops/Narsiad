nikean_artifact.1001 = {
	type = character_event

	title = nikean_artifact.1001.t
	desc = nikean_artifact.1001.desc
	theme = emperor
	override_background = bp2_study_indian
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = personality_dishonorable
	}

	immediate = {
		play_music_cue = mx_cue_touching_moment
	}

	option = {
		# Sacrifice courtiers — brutal but FAST (30 days)
		name = nikean_artifact.1001.a
		give_nickname = nick_the_bloody
		add_dread = 50
		stress_impact = {
			just = minor_stress_impact_gain
			compassionate = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		random_courtier_or_guest = {
			limit = {
				is_adult = yes
				NOT = { is_close_family_of = root }
				NOT = { is_spouse_of = root }
			}
			alternative_limit = {
				count = 3
			}
			death = {
				death_reason = death_sacrificed
				killer = root
			}
		}
		hidden_effect = {
			# Lock the decision immediately — the forge is committed
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mend_narsil_decision
			}
			set_global_variable = {
				name = flag_mend_narsil_decision
				value = root
			}
			trigger_event = {
				id = nikean_artifact.1002
				days = 30
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}

	option = {
		# Spend gold & rare materials — measured but SLOW (120 days)
		name = nikean_artifact.1001.b
		give_nickname = nick_the_blessed
		remove_treasury_or_gold = major_treasury_or_gold_value
		stress_impact = {
			sadistic = minor_stress_impact_gain
			callous = miniscule_stress_impact_gain
			greedy = miniscule_stress_impact_gain
		}
		hidden_effect = {
			# Lock the decision immediately — the forge is committed
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mend_narsil_decision
			}
			set_global_variable = {
				name = flag_mend_narsil_decision
				value = root
			}
			trigger_event = {
				id = nikean_artifact.1003
				days = 120
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_trait = greedy
			}
		}
	}

	option = {
		# Not yet — decision remains available
		name = nikean_artifact.1001.c
		add_stress = miniscule_stress_impact_gain
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}
}

nikean_gunpowder.1001 = {
	type = character_event
	title = nikean_gunpowder.1001.t
	desc = nikean_gunpowder.1001.desc
	theme = emperor
	override_background = bp1_courtyard_mena
	
	left_portrait = {
		character = root
		animation = debating
	}
	right_portrait = {
		character = scope:aide
		animation = war_over_tie
	}

	immediate = {
		if = {
			limit = { exists = cp:councillor_marshal }
			cp:councillor_steward = { save_scope_as = aide }
		}
		else_if = {
			limit = { exists = cp:councillor_steward }
			cp:councillor_chancellor = { save_scope_as = aide }
		}
		else_if = {
			limit = { any_councillor = { } }
			random_councillor = { save_scope_as = aide }
		}
		else_if = {
			limit = { vassal_count >= 1 }
			ordered_vassal = {
				order_by = martial
				save_scope_as = aide
			}
		}
		else_if = {
			limit = { any_courtier = { is_alive = yes } }
			ordered_courtier = {
				order_by = martial
				save_scope_as = aide
			}
		}
		else = {
			random_pool_character = {
				province = root.capital_province
				limit = { is_adult = yes }
				save_scope_as = aide
			}
		}
	}

	option = { #Use the city's native capabilities
		name = nikean_gunpowder.1001.a
		trigger = {
			influence >= major_influence_value
		}
		show_as_unavailable = { influence < massive_influence_value }
		save_scope_as = actor
		add_character_flag = {
			flag = took_decision_munitions_research_flag
			years = 10
		}
		begin_scheme_with_agents_effect = {
			SCHEME_TYPE = munitions_research_scheme
			TARGET_TYPE = target_title
			TARGET_SCOPE = root.capital_county
			AGENT_1 = agent_alchemist
			AGENT_2 = agent_draughtsman
			AGENT_3 = agent_drillmaster
			AGENT_4 = agent_supplier
			AGENT_5 = agent_bureaucrat_title
		}
		custom_tooltip = munitions_research_success_tooltip
		change_influence = major_influence_loss
		
		stress_impact = {
			compassionate = medium_stress_impact_gain
			calm = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_energy = 0.5
			}
			modifier = {
				factor = 0
				influence < massive_influence_value
			}
		}
	}
	option = { #Hire foreign craftsmen to manufacture it
		name = nikean_gunpowder.1001.b
		trigger = {
			treasury >= major_gold_value_static_max
		}
		show_as_unavailable = { treasury < massive_gold_value_static_max }
		save_scope_as = actor
		add_character_flag = {
			flag = took_decision_munitions_research_flag
			years = 10
		}
		begin_scheme_with_agents_effect = {
			SCHEME_TYPE = munitions_research_scheme
			TARGET_TYPE = target_title
			TARGET_SCOPE = root.capital_county
			AGENT_1 = agent_alchemist
			AGENT_2 = agent_draughtsman
			AGENT_3 = agent_drillmaster
			AGENT_4 = agent_supplier
			AGENT_5 = agent_bureaucrat_title
		}
		custom_tooltip = munitions_research_success_tooltip
		remove_treasury_or_gold = major_gold_value_static_max
		
		stress_impact = {
			compassionate = medium_stress_impact_gain
			paranoid = minor_stress_impact_gain
			calm = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_greed = -0.5
			}
			modifier = {
				factor = 0
				treasury < massive_gold_value_static_max
			}
		}
	}

	option = { #Actually... never mind
		name = nikean_gunpowder.1001.c
		remove_global_variable = flag_munitions_research_done
		remove_from_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_munitions_research_done
		}
		stress_impact = {
			sadistic = minor_stress_impact_gain
			callous = miniscule_stress_impact_gain
			ambitious = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}
}

nikean_gunpowder.1011 = {
	hidden = yes

	immediate = {
		scope:scheme = {
			scheme_owner = {
				save_scope_as = owner
				random = {
					chance = scope:scheme.scheme_success_chance
					save_scope_value_as = {
						name = scheme_successful
						value = yes
					}
				}
				if = {
					limit = { exists = scope:scheme_successful }
					trigger_event = {
						id = nikean_gunpowder.1021
						days = 1
					}
				}
				else = {
					trigger_event = {
						id = nikean_gunpowder.1022
						days = 1
					}	
				}
			}
		}
	}
}
