namespace = nars_artifacts

nars_artifacts.1001 = {
	type = character_event

	title = nars_artifacts.1001.t
	desc = nars_artifacts.1001.desc
	theme = emperor
	override_background = armory
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = personality_dishonorable
	}

	immediate = {
		play_music_cue = mx_cue_touching_moment
	}

	option = { # Hire foreign craftsmen — costly, but bloodless
		name = nars_artifacts.1001.a
		add_internal_flag = special
		save_scope_as = actor
		begin_scheme_with_agents_effect = {
				SCHEME_TYPE = mend_narsil_scheme
				TARGET_TYPE = target_title
				TARGET_SCOPE = root.capital_county
				AGENT_1 = agent_smith
				AGENT_2 = agent_alchemist
				AGENT_3 = agent_supplier
				AGENT_4 = agent_theologian
				AGENT_5 = agent_cleric
			}
		remove_treasury_or_gold = major_treasury_or_gold_value
		stress_impact = {
			sadistic = minor_stress_impact_gain
			callous = miniscule_stress_impact_gain
			greedy = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				treasury >= 750
			}
		}
	}

	option = { # Not yet — decision remains available
		name = nars_artifacts.1001.b
		add_stress = miniscule_stress_impact_gain
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				treasury < 750
			}
		}
	}
}

nars_artifacts.1011 = {
	hidden = yes

	immediate = {
		scope:scheme = {
			scheme_owner = {
				save_scope_as = owner
				random = {
					chance = scope:scheme.scheme_success_chance
					save_scope_value_as = {
						name = scheme_successful
						value = yes
					}
				}
				if = {
					limit = { exists = scope:scheme_successful }
					trigger_event = {
						id = nars_narsil.9000
					}
				}
				else = {
					trigger_event = {
						id = nars_artifacts.1022
					}	
				}
			}
		}
	}
}

nars_narsil.9000 = {
	type = character_event
	title = nars_narsil.9000.t
	desc = nars_narsil.9000.desc
	
	window = big_event_window

	theme = nars_war
	widget = { gui = event_window_widget_vfx_godray container = foreground_shader_vfx_container }

	center_portrait = {
		character = root
		camera = camera_event_center_very_large
		animation = marshal
	}

	immediate = {
		play_music_cue = mx_cue_positive_effect
		every_character_artifact = {
			limit = { has_variable = can_be_used_for_narsil } 
			destroy_artifact = this
		}
	}
	
	option = {
		name = nars_narsil.9000.a
		flavor = nars_narsil.9000.flavor
		create_narsil_effect = yes
		hidden_effect = {
			scope:scheme = { end_scheme = yes }
			scope:narsil_story = { end_story = yes }
			add_to_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:flag_mend_narsil_decision
			}
		}	
	}
}
