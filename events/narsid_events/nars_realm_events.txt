# ─────────────────────────────────────────────────────────────────
#  BIGAMY / TRIGAMY DOCTRINE EVENT
# ─────────────────────────────────────────────────────────────────

narsid_realm.1021 = {
	type = character_event

	title = narsid_realm.1021.t
	desc = narsid_realm.1021.desc
	theme = emperor
	override_background = bp2_study_indian
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = personality_rational
	}

	option = {
		# Bigamy: allows exactly TWO spouses simultaneously
		name = narsid_realm.1021.a
		custom_tooltip = narsid_realm.1021.a.bigamy_tt
		hidden_effect = {
			faith = {
				if = {
					limit = { has_doctrine = doctrine_monogamy }
					remove_doctrine = doctrine_monogamy
				}
				if = {
					limit = { has_doctrine = doctrine_concubines }
					remove_doctrine = doctrine_concubines
				}
				if = {
					limit = { has_doctrine = doctrine_polygamy }
					remove_doctrine = doctrine_polygamy
				}
				add_doctrine = doctrine_bigamy 
			}
		}
	}

	option = {
		# Trigamy: allows exactly THREE spouses simultaneously
		name = narsid_realm.1021.b
		custom_tooltip = narsid_realm.1021.b.trigamy_tt
		hidden_effect = {
			faith = {
				if = {
					limit = { has_doctrine = doctrine_monogamy }
					remove_doctrine = doctrine_monogamy
				}
				if = {
					limit = { has_doctrine = doctrine_concubines }
					remove_doctrine = doctrine_concubines
				}
				if = {
					limit = { has_doctrine = doctrine_polygamy }
					remove_doctrine = doctrine_polygamy
				}
				if = {
					limit = { has_doctrine = doctrine_bigamy }
					remove_doctrine = doctrine_bigamy
				}
				add_doctrine = doctrine_trigamy 
			}
		}
	}
}

#  HARNESS BLACK POWDER EVENT
# ─────────────────────────────────────────────────────────────────

narsid_realm.1031 = {
	type = character_event
	title = narsid_realm.1031.t
	desc = narsid_realm.1031.desc
	theme = emperor
	override_background = { reference = army_camp }
	
	left_portrait = {
		character = root
		animation = debating
	}
	right_portrait = {
		character = scope:aide
		animation = war_over_tie
	}

	immediate = {
		if = {
			limit = { exists = cp:councillor_marshal }
			cp:councillor_steward = { save_scope_as = aide }
		}
		else_if = {
			limit = { exists = cp:councillor_steward }
			cp:councillor_chancellor = { save_scope_as = aide }
		}
		else_if = {
			limit = { any_councillor = { } }
			random_councillor = { save_scope_as = aide }
		}
		else_if = {
			limit = { vassal_count >= 1 }
			ordered_vassal = {
				order_by = martial
				save_scope_as = aide
			}
		}
		else_if = {
			limit = { any_courtier = { is_alive = yes } }
			ordered_courtier = {
				order_by = martial
				save_scope_as = aide
			}
		}
		else = {
			random_pool_character = {
				province = root.capital_province
				limit = { is_adult = yes }
				save_scope_as = aide
			}
		}
	}

	option = { #Use the city's native capabilities
		name = narsid_realm.1031.a
		trigger = {
			influence >= major_influence_value
		}
		show_as_unavailable = { influence < massive_influence_value }
		save_scope_as = actor
		add_character_flag = {
			flag = took_decision_munitions_research_flag
			years = 10
		}
		begin_scheme_with_agents_effect = {
			SCHEME_TYPE = munitions_research_scheme
			TARGET_TYPE = target_title
			TARGET_SCOPE = root.capital_county
			AGENT_1 = agent_alchemist
			AGENT_2 = agent_draughtsman
			AGENT_3 = agent_drillmaster
			AGENT_4 = agent_supplier
			AGENT_5 = agent_bureaucrat_title
		}
		custom_tooltip = munitions_research_success_tooltip
		change_influence = major_influence_loss
		
		stress_impact = {
			compassionate = medium_stress_impact_gain
			calm = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_energy = 0.5
			}
			modifier = {
				factor = 0
				influence < massive_influence_value
			}
		}
	}
	option = { #Hire foreign craftsmen to manufacture it
		name = narsid_realm.1031.b
		trigger = {
			treasury >= major_gold_value_static_max
		}
		show_as_unavailable = { treasury < massive_gold_value_static_max }
		save_scope_as = actor
		add_character_flag = {
			flag = took_decision_munitions_research_flag
			years = 10
		}
		begin_scheme_with_agents_effect = {
			SCHEME_TYPE = munitions_research_scheme
			TARGET_TYPE = target_title
			TARGET_SCOPE = root.capital_county
			AGENT_1 = agent_alchemist
			AGENT_2 = agent_draughtsman
			AGENT_3 = agent_drillmaster
			AGENT_4 = agent_supplier
			AGENT_5 = agent_bureaucrat_title
		}
		custom_tooltip = munitions_research_success_tooltip
		remove_treasury_or_gold = major_gold_value_static_max
		
		stress_impact = {
			compassionate = medium_stress_impact_gain
			paranoid = minor_stress_impact_gain
			calm = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_greed = -0.5
			}
			modifier = {
				factor = 0
				treasury < massive_gold_value_static_max
			}
		}
	}
	option = { #Actually... never mind
		name = narsid_realm.1031.c
		# Refund the one-time lock since the player backed out
		hidden_effect = {
			remove_global_variable = flag_munitions_research_done
			remove_from_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_munitions_research_done
			}
		}
		stress_impact = {
			sadistic = minor_stress_impact_gain
			callous = miniscule_stress_impact_gain
			ambitious = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}
}
