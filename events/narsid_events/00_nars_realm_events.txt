# ─────────────────────────────────────────────────────────────────
#  BIGAMY / TRIGAMY DOCTRINE EVENT
# ─────────────────────────────────────────────────────────────────

narsid_realm.1021 = {
	type = character_event
	title = narsid_realm.1021.t
	desc = narsid_realm.1021.desc
	theme = emperor
	override_background = bp2_study_indian
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = personality_rational
	}

	option = {
		# Bigamy: allows exactly TWO spouses simultaneously
		name = narsid_realm.1021.a
		custom_tooltip = narsid_realm.1021.a.bigamy_tt
		hidden_effect = {
			faith = {
				if = {
					limit = { has_doctrine = doctrine_monogamy }
					remove_doctrine = doctrine_monogamy
				}
				if = {
					limit = { has_doctrine = doctrine_concubines }
					remove_doctrine = doctrine_concubines
				}
				if = {
					limit = { has_doctrine = doctrine_polygamy }
					remove_doctrine = doctrine_polygamy
				}
				add_doctrine = doctrine_bigamy 
			}
		}
	}

	option = {
		# Trigamy: allows exactly THREE spouses simultaneously
		name = narsid_realm.1021.b
		custom_tooltip = narsid_realm.1021.b.trigamy_tt
		hidden_effect = {
			faith = {
				if = {
					limit = { has_doctrine = doctrine_monogamy }
					remove_doctrine = doctrine_monogamy
				}
				if = {
					limit = { has_doctrine = doctrine_concubines }
					remove_doctrine = doctrine_concubines
				}
				if = {
					limit = { has_doctrine = doctrine_polygamy }
					remove_doctrine = doctrine_polygamy
				}
				if = {
					limit = { has_doctrine = doctrine_bigamy }
					remove_doctrine = doctrine_bigamy
				}
				add_doctrine = doctrine_trigamy 
			}
		}
	}
}
