nikean_realm.1021 = {
    type = character_event

    title = nikean_realm.1021.t
    desc = nikean_realm.1021.desc
    theme = emperor
    override_background = army_camp
    override_effect_2d = { reference = smoke }
    widget = { gui = event_window_widget_vfx_fog container = foreground_shader_vfx_container }

    left_portrait = {
        character = root
        camera = camera_event_left
        animation = interested
    }
    
    # Changed to right_portrait to avoid overlapping with the player character
    right_portrait = {
        character = scope:hoplite
        camera = camera_event_right
        animation = throne_room_one_handed_passive_1
    }

    immediate = {
        # Generate the roaming Hoplite leader
        create_character = {
            gender = male
            age = { 35 45 }
            template = nikean_template.1012
            martial >= 35
            prowess >= 45
            learning = { 25 45 }
            intrigue = { 25 45 }
            stewardship = { 25 45 }
            diplomacy = { 25 45 }
            trait = organizer
            trait = gallivanter
            trait = viking
            trait = adventurer
            save_scope_as = hoplite
        }

    }

    # Option A: Pay gold for special soldiers
    option = {
        name = nikean_realm.1021.a # "I will pay you handsomely for your spears."
        remove_treasury_or_gold = 1150
        
        spawn_army = {
			name = nikean_troop_event_name
			men_at_arms = {
				type = maa_hoplitai
				stacks = 30
            }
            location = root.location
			uses_supply = yes
			inheritable = yes
		}
        
        scope:hoplite = {
            set_employer = root
        }
    }
    option = {
        name = nikean_realm.1021.b
        
        random_list = {
            50 = {
                modifier = {
                    add = 10
                    diplomacy >= 25
                }
                modifier = {
                    add = 20
                    diplomacy >= 35
                }                
                spawn_army = {
                    name = nikean_troop_event_name
                    men_at_arms = {
                        type = maa_hoplitai
                        stacks = 30
                    }
                    location = root.location
                    uses_supply = yes
                    inheritable = yes
                }
                scope:hoplite = {
                    set_employer = root
                }
                send_interface_message = {
                    title = nikean_realm.1021.b.success_msg.t
                    desc = nikean_realm.1021.b.success_msg.desc
                    type = event_prowess_good_desc
                    left_icon = scope:hoplite
                }
            }
            50 = {
                modifier = {
                    add = 30
                    diplomacy < 15
                }
                add_prestige = medium_prestige_loss
                add_stress = medium_stress_gain
                send_interface_message = {
                    title = nikean_realm.1021.b.failure_msg.t
                    desc = nikean_realm.1021.b.failure_msg.desc
                    type = event_prowess_bad_desc
                    left_icon = scope:hoplite
                }
            }
        }
    }

    option = {
        name = nikean_realm.1021.c
        add_prestige = minor_prestige_loss
        add_stress = minor_stress_gain
    }
}
