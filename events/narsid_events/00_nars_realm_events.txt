nars_realm.1001 = {
	type = character_event

	title = nars_realm.1001.t
	desc = nars_realm.1001.desc
	theme = emperor
	override_background = armory
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = map_dismissal
	}
	option = {
		name = nars_realm.1001.a
		custom_tooltip = industrial_boom_province_tt
        add_character_modifier = {
			modifier = industrial_boom_character_modifier
			years = 10
		}
		hidden_effect = {
			every_held_title = {
				limit = { tier = tier_county } # Corrected filtering syntax
				add_county_modifier = {
					modifier = industrial_boom_province_modifier
					years = 10
				}
			}
		}
	}
}
