namespace = nikean_courtiers

nikean_courtiers.0001 = {
	hidden = yes

	trigger = {
		exists = var:number_of_invite_nikean_countdown
	}

	immediate = {
		save_temporary_scope_as = host

		# Build a list of one province per duchy in the realm
		every_realm_province = {
			if = {
				limit = {
					save_temporary_scope_as = potential_province
					NOT = {
						any_in_list = {
							list = pool_provinces
							county.duchy = scope:potential_province.county.duchy
						}
					}
				}
				add_to_temporary_list = pool_provinces
			}
		}

		# Gather Nikean pool characters with suitable prowess
		every_in_list = {
			list = pool_provinces
			save_temporary_scope_as = pool_province

			every_pool_character = {
				province = scope:pool_province
				limit = {
					prowess >= invite_nikean_courtiers_standard_value
					OR = {
						culture = { has_cultural_pillar = heritage_nars }
						has_trait = nars_trait
					}
					guest_knight_candidate_trigger = { HOST = root }
				}
				add_to_temporary_list = potential_nikean_courtiers
			}
		}

		# Pick the best candidate (weighted by how far above minimum prowess they are)
		random_in_list = {
			list = potential_nikean_courtiers
			limit = { is_alive = yes }
			weight = {
				base = 1
				modifier = {
					add = {
						value = prowess
						subtract = invite_nikean_courtiers_standard_value
						multiply = 5
					}
				}
				modifier = {
					add = {
						value = learning
						subtract = invite_nikean_courtiers_standard_value
						multiply = 5
					}
				}
			}
			save_scope_as = nikean_courtier
		}

		# If no suitable pool character exists, create a fresh Nikean warrior
		if = {
			limit = { NOT = { exists = scope:nikean_courtier } }
			create_character = {
				location = root.capital_province
				culture = root.culture
				faith = root.faith
				gender_female_chance = 50
				prowess = { invite_nikean_courtiers_standard_value invite_nikean_courtiers_upper_value }
				save_scope_as = nikean_courtier
			}
		}

		# Apply tradition bonus if culture has better_knights_from_decision
		if = {
			limit = {
				root.culture = {
					has_cultural_parameter = better_knights_from_decision
				}
			}
			hidden_effect = {
				scope:nikean_courtier = {
					blademaster_lifestyle_rank_up_effect = yes
					knight_increase_prowess_chance_effect = yes
				}
			}
		}

		# Bring the courtier to court and notify the player
		add_courtier = scope:nikean_courtier
		send_interface_message = {
			title = nikean_courtiers.0001.t
			desc = nikean_courtiers.0001.desc
			type = event_prowess_good_desc
			left_icon = scope:nikean_courtier
		}

		# Count down and either end the chain or schedule the next arrival
		change_variable = {
			name = number_of_invite_nikean_countdown
			subtract = 1
		}

		if = {
			limit = { var:number_of_invite_nikean_countdown = 0 }
			remove_variable = number_of_invite_nikean_countdown
		}
		else = {
			clear_saved_scope = nikean_courtier
			trigger_event = {
				id = nikean_courtiers.0001
				days = { 100 200 }
			}
		}
	}
}

nikean_courtiers.0011 = {
	type = character_event
	title = nikean_courtiers.0011.t
	desc = nikean_courtiers.0011.desc
	theme = emperor
	override_background = bp2_study_indian
	override_effect_2d = { reference = smoke }

		character = scope:pugilist
		animation = map_dismissal
		camera = camera_event_scheme_left
	}
	right_portrait = {
		character = scope:savant
		camera = camera_event_scheme_left
		animation = thinking
	}

	trigger = {
		exists = capital_province
	}

	immediate = {
		hidden_effect = {
			#────────────────────────────────────────────────
			# PUGILIST 
			#────────────────────────────────────────────────
			random_pool_character = {
				province = capital_province
				limit = {
					martial >= high_skill_rating
					prowess >= high_skill_rating
					learning >= high_skill_rating
					is_adult = yes
					is_alive = yes
					culture = { has_cultural_pillar = heritage_nars }
				}
				alternative_limit = {
					martial >= medium_skill_rating
					prowess >= medium_skill_rating
					learning >= medium_skill_rating
					is_adult = yes
					is_alive = yes
					culture = { has_cultural_pillar = heritage_nars }
				}
				weight = {
					base = 1
					modifier = { faith = root.faith factor = 10 }
				}
				save_scope_as = pugilist
			}
			if = {
				limit = { NOT = { exists = scope:pugilist } }
				create_character = {
					location = root.capital_province
					culture = root.culture
					faith = root.faith
					gender_female_chance = 50
					martial = { 15 30 }
					prowess = { 15 30 }
					learning = { 10 16 }
					age = { 25 50 }
					save_scope_as = pugilist
				}
			}
			scope:pugilist ?= {
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = lifestyle_blademaster } }
						add_trait = lifestyle_blademaster
					}
				}
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = strong } }
						add_trait = strong
					}
				}
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = physique_good_3 } }
						add_trait = physique_good_3
					}
				}
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = beauty_good_3 } }
						add_trait = beauty_good_3
					}
				}
			}

			#────────────────────────────────────────────────
			# SAVANT — high learning
			#────────────────────────────────────────────────
			random_pool_character = {
				province = capital_province
				limit = {
					learning >= high_skill_rating
					stewardship >= high_skill_rating
					intrigue >= high_skill_rating
					is_adult = yes
					is_alive = yes
					NOT = { scope:pugilist ?= this }
					culture = { has_cultural_pillar = heritage_nars }
				}
				alternative_limit = {
					learning >= medium_skill_rating
					stewardship >= medium_skill_rating
					intrigue >= medium_skill_rating
					is_adult = yes
					is_alive = yes
					culture = { has_cultural_pillar = heritage_nars }
					NOT = { scope:pugilist ?= this }
				}
				weight = {
					base = 1
					modifier = { faith = root.faith factor = 10 }
				}
				save_scope_as = savant
			}
			if = {
				limit = { NOT = { exists = scope:savant } }
				create_character = {
					location = root.capital_province
					culture = root.culture
					faith = root.faith
					gender_female_chance = 50
					learning = { 15 30 }
					stewardship = { 15 30 }
					intrigue = { 15 30 }
					age = { 25 50 }
					save_scope_as = savant
				}
			}
			scope:savant ?= {	
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = scholar } }
						add_trait = scholar
					}
				}
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = shrewd } }
						add_trait = shrewd
					}
				}
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = intellect_good_3 } }
						add_trait = intellect_good_3
					}
				}
				random = {
					chance = 15
					if = {
						limit = { NOT = { has_trait = beauty_good_3 } }
						add_trait = beauty_good_3
					}
				}
			}
		}
	}

	option = {
		name = nikean_courtiers.0011.a
		trigger = {
			is_playable_character = yes
			exists = scope:pugilist
		}
		pay_treasury_or_gold = {
			target = scope:pugilist
			value = major_treasury_or_gold_value
		}
		add_courtier = scope:pugilist
		ai_chance = {
			factor = 100
			modifier = {
				short_term_gold < medium_gold_value
				factor = 0
			}
		}
	}

	option = {
		name = nikean_courtiers.0011.b
		trigger = {
			is_playable_character = yes
			exists = scope:savant
		}
		pay_treasury_or_gold = {
			target = scope:savant
			value = major_treasury_or_gold_value
		}
		add_courtier = scope:savant
		ai_chance = {
			factor = 100
			modifier = {
				short_term_gold < medium_gold_value
				factor = 0
			}
		}
	}

	option = {
		name = {
			trigger = {
				NOR = {
					exists = scope:pugilist
					exists = scope:savant
				}
			}
			text = nikean_courtiers.0011.d.none
		}
		name = {
			trigger = {
				OR = {
					exists = scope:pugilist
					exists = scope:savant
				}
			}

		}
