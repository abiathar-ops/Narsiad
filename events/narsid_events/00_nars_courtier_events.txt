namespace = nikean_courtiers

nikean_courtiers.0001 = {
	hidden = yes

	trigger = {
		exists = var:number_of_invite_nikean_countdown
	}

	immediate = {
		save_temporary_scope_as = host

		# Build a list of one province per duchy in the realm
		every_realm_province = {
			if = {
				limit = {
					save_temporary_scope_as = potential_province
					NOT = {
						any_in_list = {
							list = pool_provinces
							county.duchy = scope:potential_province.county.duchy
						}
					}
				}
				add_to_temporary_list = pool_provinces
			}
		}

		# Gather Nikean pool characters with suitable prowess
		every_in_list = {
			list = pool_provinces
			save_temporary_scope_as = pool_province

			every_pool_character = {
				province = scope:pool_province
				limit = {
					culture = { has_cultural_pillar = heritage_nars }
					guest_knight_candidate_trigger = { HOST = root }
				}
				add_to_temporary_list = potential_nikean_courtiers
			}
		}

		# Pick the best candidate (weighted by how far above minimum prowess they are)
		random_in_list = {
			list = potential_nikean_courtiers
			limit = { is_alive = yes }
			weight = {
				base = 1
				modifier = {
					add = {
						value = prowess
						subtract = invite_nikean_courtiers_standard_value
						multiply = 5
					}
				}
				modifier = {
					add = {
						value = learning
						subtract = invite_nikean_courtiers_standard_value
						multiply = 5
					}
				}
			}
			save_scope_as = nikean_courtier
		}

		# If no suitable pool character exists, create a fresh Nikean warrior
		if = {
			limit = { NOT = { exists = scope:nikean_courtier } }
			create_character = {
				location = root.capital_province
				culture = root.culture
				faith = root.faith
				gender_female_chance = 50
				save_scope_as = nikean_courtier
			}
		}

		# Bring the courtier to court and notify the player
		add_courtier = scope:nikean_courtier
		send_interface_message = {
			title = nikean_courtiers.0001.t
			desc = nikean_courtiers.0001.desc
			type = event_prowess_good_desc
			left_icon = scope:nikean_courtier
		}

		# Count down and either end the chain or schedule the next arrival
		change_variable = {
			name = number_of_invite_nikean_countdown
			subtract = 1
		}

		if = {
			limit = { var:number_of_invite_nikean_countdown = 0 }
			remove_variable = number_of_invite_nikean_countdown
		}
		else = {
			clear_saved_scope = nikean_courtier
			trigger_event = {
				id = nikean_courtiers.0001
				days = { 100 200 }
			}
		}
	}
}

nikean_gunpowder.1001 = {
	type = character_event
	title = nikean_gunpowder.1001.t
	desc = nikean_gunpowder.1001.desc
	theme = emperor
	override_background = bp1_courtyard_mena
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		animation = debating
	}
	right_portrait = {
		character = scope:aide
		animation = war_over_tie
	}

	immediate = {
		if = {
			limit = { exists = cp:councillor_marshal }
			cp:councillor_steward = { save_scope_as = aide }
		}
		else_if = {
			limit = { exists = cp:councillor_steward }
			cp:councillor_chancellor = { save_scope_as = aide }
		}
		else_if = {
			limit = { any_councillor = { } }
			random_councillor = { save_scope_as = aide }
		}
		else_if = {
			limit = { vassal_count >= 1 }
			ordered_vassal = {
				order_by = martial
				save_scope_as = aide
			}
		}
		else_if = {
			limit = { any_courtier = { is_alive = yes } }
			ordered_courtier = {
				order_by = martial
				save_scope_as = aide
			}
		}
		else = {
			random_pool_character = {
				province = root.capital_province
				limit = { is_adult = yes }
				save_scope_as = aide
			}
		}
	}

	option = { #Use the city's native capabilities
		name = nikean_gunpowder.1001.a
		trigger = {
			influence >= major_influence_value
		}
		show_as_unavailable = { influence < massive_influence_value }
		save_scope_as = actor
		add_character_flag = {
			flag = took_decision_munitions_research_flag
			years = 10
		}
		begin_scheme_with_agents_effect = {
			SCHEME_TYPE = munitions_research_scheme
			TARGET_TYPE = target_title
			TARGET_SCOPE = root.capital_county
			AGENT_1 = agent_alchemist
			AGENT_2 = agent_draughtsman
			AGENT_3 = agent_drillmaster
			AGENT_4 = agent_supplier
			AGENT_5 = agent_bureaucrat_title
		}
		custom_tooltip = munitions_research_success_tooltip
		change_influence = major_influence_loss
		
		stress_impact = {
			compassionate = medium_stress_impact_gain
			calm = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_energy = 0.5
			}
			modifier = {
				factor = 0
				influence < massive_influence_value
			}
		}
	}
	option = { #Hire foreign craftsmen to manufacture it
		name = nikean_gunpowder.1001.b
		trigger = {
			treasury >= major_gold_value_static_max
		}
		show_as_unavailable = { treasury < massive_gold_value_static_max }
		save_scope_as = actor
		add_character_flag = {
			flag = took_decision_munitions_research_flag
			years = 10
		}
		begin_scheme_with_agents_effect = {
			SCHEME_TYPE = munitions_research_scheme
			TARGET_TYPE = target_title
			TARGET_SCOPE = root.capital_county
			AGENT_1 = agent_alchemist
			AGENT_2 = agent_draughtsman
			AGENT_3 = agent_drillmaster

			AGENT_5 = agent_bureaucrat_title
		}
		custom_tooltip = munitions_research_success_tooltip
		remove_treasury_or_gold = major_gold_value_static_max
		
		stress_impact = {
			compassionate = medium_stress_impact_gain
			paranoid = minor_stress_impact_gain
			calm = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain

				ai_greed = -0.5

				treasury < massive_gold_value_static_max

	}
