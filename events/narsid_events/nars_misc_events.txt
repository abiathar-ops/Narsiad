# ─────────────────────────────────────────────────────────────────
#  NARSIL REFORGING EVENTS
#  1001 → choice (sacrifice or gold)
#  1002 → blood ritual result  (DELAY: 30 days — faster, brutal)
#  1003 → gold ritual result   (DELAY: 180 days — slower, costly)
#  1004 → Narsíl reforged (fanfare)
# ─────────────────────────────────────────────────────────────────
namespace = narsid_artifact

narsid_artifact.1001 = {
	type = character_event

	title = narsid_artifact.1001.t
	desc = narsid_artifact.1001.desc
	theme = emperor
	override_background = bp2_study_indian
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		camera = camera_center_large
		animation = personality_dishonorable
	}

	immediate = {
		play_music_cue = mx_cue_touching_moment
	}

	option = {
		# Sacrifice courtiers — brutal but FAST (30 days)
		name = narsid_artifact.1001.a
		give_nickname = nick_the_bloody
		add_dread = 50
		stress_impact = {
			just = minor_stress_impact_gain
			compassionate = miniscule_stress_impact_gain
			content = miniscule_stress_impact_gain
		}
		hidden_effect = {
			trigger_event = {
				id = narsid_artifact.1002
				days = 60
			}
		}
		random_courtier_or_guest = {
			limit = {
				is_adult = yes
				NOT = { is_close_family_of = root }
				NOT = { is_spouse_of = root }
			}
			alternative_limit = {
				count = 3
			}
			death = {
				death_reason = death_sacrificed
				killer = root
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}

	option = {
		# Spend gold & rare materials — measured but SLOW (120 days)
		name = narsid_artifact.1001.b
		give_nickname = nick_the_blessed
		remove_treasury_or_gold = major_treasury_or_gold_value
		hidden_effect = {
			# ── DELAY ADDED: 120 days ──
			trigger_event = {
				id = narsid_artifact.1003
				days = 120
			}
		}
		stress_impact = {
			sadistic = minor_stress_impact_gain
			callous = miniscule_stress_impact_gain
			greedy = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}

	option = {
		name = narsid_artifact.1001.c
		add_stress = miniscule_stress_impact_gain
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_trait = compassionate
					has_trait = calm
					has_trait = content
				}
			}
		}
	}
}
