		character = root

			}
		}
	}

	option = {
		name = narsid_artifact.1001.b #Buy Blood and Wait

		add_piety = 275
		remove_treasury_or_gold = major_treasury_or_gold_value
		hidden_effect = {
			trigger_event = narsid_artifact.1003
		}
	}
