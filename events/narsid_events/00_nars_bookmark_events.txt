nikean_bookmark.0021 = {
	type = character_event
	title = nikean_bookmark.0021.t
	desc = {
		desc = nikean_bookmark.0021.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = {
					var:previous_holder = flag:subject
				}
				desc = nikean_bookmark.0021.desc.subject
			}
		}
	}
			
	theme = nomads
	left_portrait = {
		character = root
    	animation = jockey_idle
  		camera = camera_event_horse_left
	}
	right_portrait = {
		character = scope:herder
		animation = obsequious_bow
	}
	lower_right_portrait = {
		trigger = { exists = scope:nomad }
		character = scope:nomad
	}
	
	immediate = {
		set_variable = adventurer_to_nomad_tributary_var
		location.county.holder = { save_scope_as = herder }
		if = {
			limit = {
				scope:herder = {
					is_independent_ruler = yes
					is_tributary = no
				}
			}
			set_variable = {
				name = previous_holder
				value = flag:independent
			}
		}
		else = {
			set_variable = {
				name = previous_holder
				value = flag:subject
			}
			if = {
				limit = { exists = scope:herder.suzerain }
				scope:herder.suzerain = { save_scope_as = nomad }
			}
			else = {
				scope:herder.liege = { save_scope_as = nomad }
			}
		}
	}
	
	option = { # Insult
		name = nikean_bookmark.0021.a
        custom_tooltip = nikean_bookmark.0021.a.tt
        add_prestige_level = 1
        add_prestige = 75
		stress_impact = {
			craven = massive_stress_impact_gain
		}
        trigger_event = { id = nikean_bookmark.0022 days = 7 }
		ai_chance = {
			base = 100
			modifier = {
				has_trait = craven
				factor = 0
			}
		}
	}
	
	option = { # Refuse to become Tributary
		name = nikean_bookmark.0021.b
        add_prestige = 25
		stress_impact = {
			craven = major_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = {
				has_trait = craven
				factor = 0
			}
		}
	}
	
	after = {
		bookmark_ilione_nomadic_event_effect = yes
		remove_variable = previous_holder
		remove_variable = previous_provisions
		remove_variable = adventurer_to_nomad_tributary_var
	}
}

nikean_bookmark.1003 = {
    type = character_event
    title = nikean_bookmark.1003.t   
    desc = nikean_bookmark.1003.desc
    
    theme = emperor
    override_background = temple
    
    left_portrait = {
        character = root
        animation = war_defender
        camera = camera_center_large
    }
    
    right_portrait = {
        character = scope:event_messenger
        animation = throne_room_messenger_3
        camera = camera_event_center_away_pointing_left_fov25
    }
    
    lower_right_portrait = {
        character = scope:rival_1
        animation = marshal
    }
    
    lower_center_portrait = {
        character = scope:rival_2
        animation = marshal_dagger
    }
    
    immediate = {
        if = {
            limit = { root = { has_title = title:d_laamp_nars_1001 } }
            title:d_laamp_nars_1002.holder = { save_scope_as = rival_1 }
            title:d_laamp_nars_1003.holder = { save_scope_as = rival_2 }
        }
        else_if = {
            limit = { root = { has_title = title:d_laamp_nars_1002 } }
            title:d_laamp_nars_1001.holder = { save_scope_as = rival_1 }
            title:d_laamp_nars_1003.holder = { save_scope_as = rival_2 }
        }
        else_if = {
            limit = { root = { has_title = title:d_laamp_nars_1003 } }
            title:d_laamp_nars_1001.holder = { save_scope_as = rival_1 }
            title:d_laamp_nars_1002.holder = { save_scope_as = rival_2 }
        }

        random_courtier = {
            limit = {
                is_adult = yes
                is_imprisoned = no
                is_capable = yes
                NOT = { this = root }
                NOT = { this = scope:rival_1 }
                NOT = { this = scope:rival_2 }
                NOT = { is_spouse_of = root }
            }
            save_scope_as = event_messenger
        }
        
        # Fallback Messenger
        if = {
            limit = { 
                NOT = { exists = scope:event_messenger } 
            }
            create_character = {
                culture = root.culture
                faith = root.faith
                gender_female_chance = 50
                age = { 18 25 }
                save_scope_as = event_messenger
                
                after_creation = {
                    add_trait = nars_naturalis
                    # Flag them for cleanup
                    add_character_flag = nars_temp_messenger_cleanup
                }
            }
        }
    }
    
    option = {
        name = nikean_bookmark.1003.a
        if = {
            limit = { has_title = title:d_laamp_nars_1001 }
            add_prestige_level = 1
        }   
        if = {
            limit = { has_title = title:d_laamp_nars_1002 }
            add_prestige_level = 1
        }   
        if = {
            limit = { has_title = title:d_laamp_nars_1003 }
            add_prestige_level = 1
        }   
    }

    option = {
        name = nikean_bookmark.1003.b
    }

    after = {
        if = {
            limit = { 
                exists = scope:event_messenger
                scope:event_messenger = { has_character_flag = nars_temp_messenger_cleanup } 
            }
            hidden_effect = {
                scope:event_messenger = {
                    death = { death_reason = death_vanished }
                }
            }
        }
    }
}
