story_nars_naturalis = {
    visible = yes

    icon = {
        reference = "gfx/interface/icons/artifact/artifact_icons_constantine.dds"
    }

    visualization = {
        character = {
            variable_name = "nars_char"
            label = "NARS_CHAR_LABEL"
        }
        
        # This will now display correctly
        custom_string_key = NARS_NATURALIS_STORY_PRACTICAL_INFORMATION_STRING

        basic_counter = {
            variable_name = "narsokratia_value"
            tooltip = "NARSOKRATIA_COUNTER_TOOLTIP"
            min = 0
            max = 5
            label = "NARSOKRATIA_VALUE_LABEL"
            min_label = "NARSOKRATIA_VALUE_MIN_LABEL"
            max_label = "NARSOKRATIA_VALUE_MAX_LABEL"
        }
    }
    
    on_setup = {
        # Bind the root character to the story variable
        story_owner = { 
            save_scope_as = nars_char
        }
        set_variable = {
            name = nars_char
            value = scope:nars_char
        }
        set_variable = {
			name = narsokratia_value
			value = 0
		}
        # Calculate rank immediately directly on the story scope
        nars_update_narsokratia_value_effect = yes
    }

    effect_group = { #narsscion
		days = { 100 200 }

		trigger = {
			story_owner = {
				is_available = yes
				is_alive = yes
				is_adult = yes
                top_liege = this
                has_trait = nars_scion
			}
		}

		first_valid = {
			triggered_effect = {
				trigger = { 
                    always = yes
                }
				effect = {
					story_owner = {
						trigger_event = {
							on_action = ongoing_nars_scion_events
						}
					}
				}
			}
		}
	}

    on_owner_death = {
        end_story = yes
    }
}
