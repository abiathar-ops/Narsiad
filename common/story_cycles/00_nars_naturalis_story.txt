	visible = yes

	icon = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/icons/artifact/artifact_icons_constantine.dds"
	}

	visualization = {
		# traits = {
		# 	nars_naturalis
		# }
		custom_string_key = "NARS_NATURALIS_STORY_PRACTICAL_INFORMATION_STRING"
		basic_counter = {
			variable_name = "narsokratia_rank"
			min = 0
			max = 4
			label = "NARSOKRATIA_VALUE"
		}
	}

	on_setup = {
		set_variable = {
			name = nars_naturalis_active
			value = story_owner

	effect_group = {
		days = { 100 200 }

		trigger = {
			story_owner = {

			}
		}

		first_valid = {
			triggered_effect = {
				trigger = { always = yes }
				effect = {
					save_scope_as = story_scope
					story_owner = {
						trigger_event = {
							on_action = ongoing_nars_naturalis_owner_events
						}
					}
				}
			}
		}
	}
