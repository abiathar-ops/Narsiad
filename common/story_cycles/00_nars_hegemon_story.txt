story_nars_hegemon = {
    visible = yes

    icon = {
        reference = "gfx/interface/icons/artifact/artifact_icons_constantine.dds"
    }

    background = {
        reference = "gfx/interface/illustrations/event_scenes/nars_war.dds"
    }

    visualization = {
        # character = {
        #     variable_name = "nars_char"
        #     label = "NARS_CHAR_LABEL"
        # }
        custom_string_key = "NIKATOR_STORY_PRACTICAL_INFORMATION_STRING"
    }

    on_setup = {
        story_owner = { 
            save_scope_as = nars_char
        }
        set_variable = {
            name = nars_char
            value = scope:nars_char
        }
    }
    
    # 1. Ongoing Events
    effect_group = { 
        days = { 180 420 }

        trigger = {
            story_owner = {
                is_available = yes
                is_alive = yes
                is_adult = yes
                top_liege = this
                has_trait = nars_hegemon
                NOT = { has_trait = conqueror }
            }
        }

        first_valid = {
            triggered_effect = {
                trigger = { always = yes }
                effect = {
                    story_owner = {
                        trigger_event = {
                            on_action = ongoing_nars_hegemon_events
                        }
                    }
                }
            }
        }
    }

    # 2. AI Pacing Mechanics
    effect_group = {
        days = { 30 60 }

        triggered_effect = {
            trigger = {
                story_owner = { is_ai = yes }
            }
            effect = {
                story_owner = {
                    save_scope_as = root
                    ai_conqueror_yearly_effect = yes
                    if = { 
                        limit = {
                            OR = {
                                top_liege != this
                                is_landed = no
                                highest_held_title_tier <= tier_county
                                is_tributary = yes
                            }
                            is_ai = yes
                        }
                        # FIXED: Triggers custom event, not vanilla
                        trigger_event = nars_hegemon.0002
                    }
                }
            }
        }
    }

    # 3. Cycle End Conditions (The 6 Triggers)
    effect_group = { 
        days = { 900 1800 }
    
        triggered_effect = {
            trigger = {
                story_owner = {
                    OR = {
                        AND = {
                            exists = var:lost_wars
                            var:lost_wars >= 3
                        }
                        is_ideal_selian = no
                        days_of_continuous_peace >= 3650 
                        is_independent_ruler = no
                        has_trait = incapable
                        highest_held_title_tier <= tier_duchy
                    }
                }
            }
            effect = {
                story_owner = {
                    trigger_event = nars_hegemon.0002
                }
                end_story = yes
            }
        }
    }

    on_owner_death = {
		story_owner = {
			if = {
				limit = {
					primary_heir ?= {
						is_physically_able_adult = yes
						OR = {
							ai_has_cautious_personality = no
							is_ai = no
						}
					}
				}
				primary_heir ?= {
					add_trait = nars_hegemon
					create_story = story_nars_hegemon
				}
			}
			else = {
				every_heir = {
					limit = {
						is_physically_able_adult = yes
						OR = {
							ai_has_cautious_personality = no
							is_ai = no
						}
					}
					add_trait = nars_hegemon
					create_story = story_nars_hegemon
				}
			}
		}
		end_story = yes
	}

    on_end = {
        debug_log = "Nikátōr story cycle ended on:"
        debug_log_date = yes
        # Removed event trigger here so it doesn't fire upon natural death
        story_owner = {
			if = {
				limit = {
					is_alive = yes
				}
				remove_trait = nars_hegemon
			}
		}
    }
}
