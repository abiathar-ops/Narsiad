story_nars_conqueror = {
    visible = yes

	}

    visualization = {
        basic_counter = {
            variable_name = "narsokratia_value"
            tooltip = "NARSOKRATIA_COUNTER_TOOLTIP"
            min = 0
            max = 5
            label = "NARSOKRATIA_VALUE_LABEL"
            min_label = "NARSOKRATIA_VALUE_MIN_LABEL"
            max_label = "NARSOKRATIA_VALUE_MAX_LABEL"
        }
    }

        set_variable = {
			name = narsokratia_value
			value = 0
		}

        nars_update_narsokratia_value = yes
    }

    effect_group = { #narsscion
		days = { 100 200 }

		trigger = {
			story_owner = {
				is_available = yes

				is_adult = yes
                top_liege = this
                has_trait = nars_conqueror
			}
		}

		first_valid = {
			triggered_effect = {
				trigger = { 
                    always = yes
                }
				effect = {
					story_owner = {
						trigger_event = {
							on_action = ongoing_nars_conqueror_events
						}
					}
				}
			}
		}
	}

    on_owner_death = {
        end_story = yes
    }

    on_end = {
		debug_log = "Nikátōr story cycle ended on:"
