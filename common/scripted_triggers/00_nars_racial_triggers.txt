is_nikean = {
    or = {
        culture = { has_cultural_pillar = heritage_nars }
        has_ethnicity = nikean
    }
}

is_ethnically_nikean = {
	has_ethnicity = nikean
}

has_nikean_blood = {
    has_trait = nars_naturalis
}

is_orphic = {
   faith.religion = {
		is_in_family = rf_nikean
	}
}

is_considered_orphic = {
    custom_tooltip = {
        text = is_considered_orphic
        faith.religion = {
	       	is_in_family = rf_nikean
	    }
    }
}

is_narsocratic = {
   culture = {
        has_cultural_pillar = heritage_nars
   }
   or = {
        highest_held_title_tier >= tier_kingdom
        has_title = title:h_narsokratia
   }
}

is_narsocratic_e_h = {
    highest_held_title_tier >= tier_empire
}

is_narses_scion = {
	AND = {
		has_trait = nars_scion
		is_nikean = yes
        is_orphic = yes
	}
}
