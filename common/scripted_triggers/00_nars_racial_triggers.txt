is_selian = {
    or = {
        culture = { has_cultural_pillar = heritage_nars }
        has_ethnicity = selianes
    }
}

is_ethnically_selian = {
	has_ethnicity = selianes
}

has_selian_blood = {
    has_trait = nars_naturalis
}

is_archaic = {
    faith.religion = {
		is_in_family = rf_selian
	}
}

is_considered_archaic = {
    custom_tooltip = {
        text = is_considered_archaic
        faith.religion = {
	       	is_in_family = rf_selian
	    }
    }
}

is_ideal_selian = {
    culture = { has_cultural_pillar = heritage_nars }
    has_ethnicity = selianes
    faith.religion = {
		is_in_family = rf_selian
	}
}

is_narsocratic = {
   culture = {
        has_cultural_pillar = heritage_nars
   }
   or = {
        highest_held_title_tier >= tier_kingdom
        has_title = title:h_narsokratia
   }
}

is_narsocratic_e_h = {
    highest_held_title_tier >= tier_empire
}

is_narses_scion = {
	AND = {
		has_trait = nars_scion
		is_selian = yes
        is_archaic = yes
	}
}
