recruit_nars_court_has_room_trigger = {
	OR = {
		is_ai = no	
		any_courtier = {
			count < 20
		}
	}
}

valid_for_selian_conversion_trigger = {
	# The character (the land's holder) must differ from root in BOTH faith and culture.
	NOR = {
		faith = root.faith
		culture = root.culture
	}
	is_at_war = no
	NOR = {
		has_character_flag = do_not_disturb_preachers
		primary_title = { has_variable = do_not_disturb_preachers }
	}
}

nars_faith_needs_bribe_trigger = {
	nars_faith_conversion_resistance_value >= 15
}

nars_faith_scheme_has_targets_trigger = {
	any_sub_realm_title = {
		tier = tier_county
		NOT = { faith = root.faith }
		holder = {
			valid_for_selian_conversion_trigger = yes
		}
	}
}

nars_is_ruler_scope_available = {
	county = {
		trigger_if = {
			limit = { exists = scope:ruler }
			holder = scope:ruler
		}
		trigger_else = {
			holder = $HOLDER$
		}
	}
}

is_barony_valid_for_aretaic_school = {
	nars_is_ruler_scope_available = {
		HOLDER = $HOLDER$
	}
	title_province = {
        has_special_building_slot = no
    }
	is_leased_out = no
	title_province = {
		NOT = { has_building_or_higher = aretaic_school_01 }
	}
}
