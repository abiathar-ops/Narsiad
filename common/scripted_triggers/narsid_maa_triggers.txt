# Basic Triggers

can_recruit_elite_maa = {
  OR = {
    highest_held_title_tier >= tier_duchy
    AND = {
      is_independent_ruler = yes
    }
  }
}

can_recruit_guard_maa = {
  AND = {
  is_independent_ruler = yes
  highest_held_title_tier >= tier_duchy
  }
}

# Culture Triggers

narsid_maa_trigger = { # Handles access to Narsid units
	culture = { has_cultural_pillar = heritage_nars }
}

can_recruit_maa_01_trigger = {
	AND = {
		narsid_maa_trigger = yes
		culture = { 
			has_innovation = innovation_narsid_maa_01
		}
	}
}

# ─────────────────────────────────────────────────────────────────
#  Mástiges recruitment: available either via the late-medieval
#  innovation OR after completing the Harness Black Powder decision
#  (which sets the global variable mastiges_research_complete).
# ─────────────────────────────────────────────────────────────────
can_recruit_maa_02_trigger = {
	AND = {
		narsid_maa_trigger = yes
		OR = {
			culture = { has_innovation = innovation_narsid_maa_03 }
			exists = global_var:mastiges_research_complete
		}
	}
}

#### MAA ART TRIGGERS ####
narsid_maa_art_trigger = {

}
