heritage_nars = {
	type = heritage
	is_shown = {
		heritage_is_shown_trigger = {
			HERITAGE = heritage_nars
		}
	}
	audio_parameter = nikean
}

language_nars = {
	type = language
	is_shown = {
		language_is_shown_trigger = {
			LANGUAGE = language_nars
		}
		scope:character = { is_nikean = yes }
	}
	ai_will_do = {
		value = 10
		if = {
			limit = { has_cultural_pillar = language_nars }
			multiply = 100
		}
	}
	
	color = greek
}
