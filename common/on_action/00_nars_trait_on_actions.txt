# Hook into the vanilla on_actions
on_game_start_after_lobby = {
    on_actions = { 
        nars_seed_nikean_trait_on_start 
        nars_intro_on_start
        nars_laamp_on_start
    }
}


on_birth_child = {
    on_actions = { 
        nars_add_nikean_trait_on_birth 
    }
}

# Define the custom actions triggered by the hooks above

nars_intro_on_start = {
    effect = {
        every_player = {
            if = {
                limit = { 
                    is_ethnically_nikean = yes 
                    is_alive = yes
                }
			    trigger_event = { id = nikean_intro.1001 days = 1 }
		    }
        }
    } 
}

nars_seed_nikean_trait_on_start = {
    effect = {
        every_living_character = {
            limit = { 
                # Target the specific culture. 
                # Alternatively, use 'has_cultural_heritage = heritage_nars' if it should apply to diverged cultures too.
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_add_nikean_trait_on_birth = {
    effect = {
        # In the 'on_birth_child' on_action, the ROOT scope is the newborn child.
        if = {
            limit = {
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_laamp_on_start = {
    effect = {
        if = {
            limit = { has_ep3_dlc_trigger = yes }
            
            # Ēleîos
            character:nars_0001 ?= {
                if = {
                    limit = { 
                        is_alive = yes 
                        is_landless_adventurer = yes
                    }
                    # Everything below happens ONLY if the limits above are true
                    set_variable = {
                        name = number_of_invite_nikean_countdown
                        value = 7
                        days = 1000
                    }
                    trigger_event = { 
                        id = nikean_courtiers.0001 
                        days = { 15 30 } 
                    }
                    add_realm_law_skip_effects = camp_purpose_mercenaries
                }
            }
            
            # Iliónē
            character:nars_0002 ?= {
                 if = {
                    limit = { 
                        is_alive = yes 
                        is_landless_adventurer = yes
                    }
                    set_variable = {
                        name = number_of_invite_nikean_countdown
                        value = 7
                        days = 1000
                    }
                    trigger_event = { 
                        id = nikean_courtiers.0001 
                    }
                    add_realm_law_skip_effects = camp_purpose_mercenaries
                }
            }
            
            # Stratō
            character:nars_0003 ?= {
                if = {
                    limit = { 
                        is_alive = yes 
                        is_landless_adventurer = yes
                    }
                    set_variable = {
                        name = number_of_invite_nikean_countdown
                        value = 7
                        days = 1000
                    }
                    trigger_event = { 
                        id = nikean_courtiers.0001 
                        days = { 15 30 } 
                    }
                    add_realm_law_skip_effects = camp_purpose_mercenaries
                }
            }
        }
    }
}
