on_game_start_after_lobby = {
    on_actions = { 
        nars_seed_nikean_trait_on_start 
        nars_intro_on_start
        nars_laamp_on_start
    }
}


on_birth_child = {
    on_actions = { 
        nars_add_nikean_trait_on_birth 
    }
}

nars_intro_on_start = {
    effect = {
        every_player = {
            limit = { 
                is_ethnically_nikean = yes 
                is_alive = yes
            }
            trigger_event = { id = nikean_intro.1001 days = 1 }
        }

        character:nars_0001 ?= {
            if = {
                limit = { is_alive = yes }              
                trigger_event = nikean_courtiers.0002 
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }

        character:nars_0002 ?= {
            if = {
                limit = { is_alive = yes }
                
                trigger_event = nikean_courtiers.0002 
                
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }

        character:nars_0003 ?= {
            if = {
                limit = { is_alive = yes }
                
                trigger_event = nikean_courtiers.0002 
                
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }
    } 
}

nars_seed_nikean_trait_on_start = {
    effect = {
        every_living_character = {
            limit = { 
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_add_nikean_trait_on_birth = {
    effect = {
        if = {
            limit = {
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}
