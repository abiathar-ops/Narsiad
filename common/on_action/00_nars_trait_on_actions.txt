# Hook into the vanilla on_actions
on_game_start_after_lobby = {
    on_actions = { 
        nars_seed_nikean_trait_on_start 
        nars_intro_on_start
        nars_laamp_on_start
    }
}


on_birth_child = {
    on_actions = { 
        nars_add_nikean_trait_on_birth 
    }
}

# Define the custom actions triggered by the hooks above

nars_intro_on_start = {
    effect = {
        # 1. Any player who is Nikean gets the intro (Fires ONCE for the player)
        every_player = {
            limit = { 
                is_ethnically_nikean = yes 
                is_alive = yes
            }
            trigger_event = { id = nikean_intro.1001 days = 1 }
        }

        # 2. Ēleîos Setup
        character:nars_0001 ?= {
            if = {
                limit = { is_alive = yes }
                
                # Everyone (Player or AI) gets the 5 free courtiers
                trigger_event = nikean_courtiers.0002 
                
                # ONLY give the intro event here if they are AI (prevents player duplicate)
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }

        # 3. Iliónē Setup
        character:nars_0002 ?= {
            if = {
                limit = { is_alive = yes }
                
                trigger_event = nikean_courtiers.0002 
                
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }

        # 4. Stratō Setup
        character:nars_0003 ?= {
            if = {
                limit = { is_alive = yes }
                
                trigger_event = nikean_courtiers.0002 
                
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }
    } 
}

nars_seed_nikean_trait_on_start = {
    effect = {
        every_living_character = {
            limit = { 
                # Target the specific culture. 
                # Alternatively, use 'has_cultural_heritage = heritage_nars' if it should apply to diverged cultures too.
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_add_nikean_trait_on_birth = {
    effect = {
        # In the 'on_birth_child' on_action, the ROOT scope is the newborn child.
        if = {
            limit = {
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_laamp_on_start = {
    effect = {
        if = {
            limit = { has_ep3_dlc_trigger = yes }
            
            # Ēleîos
            character:nars_0001 ?= {
                if = {
                    limit = { 
                        is_alive = yes 
                        is_landless_adventurer = yes
                    }
                    add_realm_law_skip_effects = camp_purpose_mercenaries
                }
            }
            
            # Iliónē
            character:nars_0002 ?= {
                 if = {
                    limit = { 
                        is_alive = yes 
                        is_landless_adventurer = yes
                    }
                    add_realm_law_skip_effects = camp_purpose_mercenaries
                }
            }
            
            # Stratō
            character:nars_0003 ?= {
                if = {
                    limit = { 
                        is_alive = yes 
                        is_landless_adventurer = yes
                    }
                    add_realm_law_skip_effects = camp_purpose_mercenaries
                }
            }
        }
    }
}
