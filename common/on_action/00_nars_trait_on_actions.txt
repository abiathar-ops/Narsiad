on_character_created = {
    effect = {
        if = {
            limit = {
                is_ethnically_nikean = yes
                NOT = { has_trait = nars_naturalis }
            }
            add_trait = nars_naturalis
            create_story = story_nars_naturalis
            nars_naturalis_set_tier_xp_effect = yes
        }
    }
}

nars_update_rank_on_action = {
    effect = {
        # ROOT is the character. Scope to their story and fire the update!
        every_owned_story = {
            limit = { story_type = story_nars_naturalis }
            nars_update_narsokratia_value = yes
        }
    }
}

random_yearly_everyone_pulse = {
    on_actions = { nars_one_time_maintenance_on_action }
}

on_trait_added = {
    effect = {
        if = {
            limit = { has_trait = nars_naturalis }
            nars_naturalis_set_tier_xp_effect = yes
        }
    }
}

nars_intro_on_start = {
    effect = {
        every_player = {
            limit = { 
                is_nikean = yes 
                is_alive = yes
            }
            trigger_event = { id = nikean_intro.1001 days = 1 }
            while = {
                count = 3
                trigger_event = { id = nikean_courtiers.0002 }
            }
            add_prestige_level = 2
            add_piety_level = 1
            add_gold = { 705 1000 }
            add_prestige = { 705 1000 }
        }
    } 
}

nars_add_nikean_trait_on_start = {
    effect = {
        every_living_character = {
            limit = { 
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
            create_story = story_nars_naturalis
            nars_naturalis_set_tier_xp_effect = yes
        }
    }
}

nars_add_nikean_trait_on_birth = {

            }

        }

}

nars_laamp_on_start = {
    effect = {
        every_domicile = {
            limit = {
                owner = {
                    is_alive = yes
                    is_landless_adventurer = yes
                    is_ethnically_nikean = yes
                }
            }
            nars_laamp_camp_setup_effect = yes
        }
    }    
}

nars_naturalis_sync_all_on_start = {
    on_actions = {
        nars_update_rank_on_action
    }
    effect = {
        every_living_character = {
            limit = { has_trait = nars_naturalis }
            nars_naturalis_set_tier_xp_effect = yes
        }
    }
}
