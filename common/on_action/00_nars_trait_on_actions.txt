on_game_start_after_lobby = {
    on_actions = { 
        nars_seed_nikean_trait_on_start 
        nars_intro_on_start
        nars_laamp_on_start
    }
}


on_birth_child = {
    on_actions = { 
        nars_add_nikean_trait_on_birth 
    }
}

nars_intro_on_start = {
    effect = {
        every_player = {
            limit = { 
                is_ethnically_nikean = yes 
                is_alive = yes
            }
            trigger_event = { id = nikean_intro.1001 days = 1 }
        }

        character:nars_0001 ?= {
            if = {
                limit = { is_alive = yes }              
                trigger_event = nikean_courtiers.0002 
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }

        character:nars_0002 ?= {
            if = {
                limit = { is_alive = yes }
                
                trigger_event = nikean_courtiers.0002 
                
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }

        character:nars_0003 ?= {
            if = {
                limit = { is_alive = yes }
                
                trigger_event = nikean_courtiers.0002 
                
                if = {
                    limit = { is_ai = yes }
                    trigger_event = { id = nikean_intro.1001 days = 1 }
                }
            }
        }
    } 
}

nars_seed_nikean_trait_on_start = {
    effect = {
        every_living_character = {
            limit = { 
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_add_nikean_trait_on_birth = {
    effect = {
        if = {
            limit = {
                is_ethnically_nikean = yes
                NOT = { has_nikean_blood = yes }
            }
            add_trait = nars_naturalis
        }
    }
}

nars_laamp_on_start = {
    effect = {
        # --- PHASE 1: SIBLINGS & MAIN CAMP SETUP ---
        if = {
            limit = { has_ep3_dlc_trigger = yes }
            
            character:nars_0001 ?= {
                add_realm_law_skip_effects = camp_purpose_mercenaries
            }
            character:nars_0002 ?= {
                add_realm_law_skip_effects = camp_purpose_mercenaries
            }
            character:nars_0003 ?= {
                add_realm_law_skip_effects = camp_purpose_mercenaries
            }
        }

        # --- PHASE 2: AUTOMATED CAMPS UPGRADE SEEDING ---
        every_domicile = {
            # Check the domicile's OWNER to see if they are a living Nikean adventurer
            limit = {
                owner = {
                    is_alive = yes
                    is_landless_adventurer = yes
                    is_ethnically_nikean = yes
                }
            }

            # Force basic Headquarters upgrades immediately
            add_domicile_building = camp_main_02
            add_domicile_building = camp_main_03
            
            # --- TIER 1 ROLLS ---
            random_list = {
                10 = {
                    # Roll for Supply Tent tree
                    add_domicile_building = supply_tent_01
                    add_domicile_building = supply_tent_02
                    random_list = {
                        10 = { add_domicile_building = supply_tent_sutler }
                        10 = { add_domicile_building = supply_tent_mender }
                        10 = { add_domicile_building = supply_tent_smithy }
                        10 = { add_domicile_building = supply_tent_arsenal }
                    }
                }
                10 = {
                    # Roll for Barber Tent tree
                    add_domicile_building = barber_tent_01
                    add_domicile_building = barber_tent_02
                    random_list = {
                        10 = { add_domicile_building = barber_tent_surgeons_tools }
                        10 = { add_domicile_building = barber_tent_dentists_tools }
                        10 = { add_domicile_building = barber_tent_torturers_tools }
                    }
                }
                10 = {
                    # Roll for Baggage Train tree
                    add_domicile_building = baggage_train_01
                    add_domicile_building = baggage_train_02
                    random_list = {
                        10 = { add_domicile_building = baggage_train_ample_steeds }
                        10 = { add_domicile_building = baggage_train_porters }
                        10 = { add_domicile_building = baggage_train_trackers }
                        10 = { add_domicile_building = baggage_train_kennel }
                    }
                }
            }
            
            # --- TIER 2 ROLLS ---
            random_list = {
                10 = {
                    # Roll for Mess Tent tree
                    add_domicile_building = mess_tent_01
                    add_domicile_building = mess_tent_02
                    random_list = {
                        10 = { add_domicile_building = mess_tent_herbalists }
                        10 = { add_domicile_building = mess_tent_brewers }
                        10 = { add_domicile_building = mess_tent_curers }
                        10 = { add_domicile_building = mess_tent_bakers }
                        10 = { add_domicile_building = mess_tent_cooks }
                    }
                }
                10 = {
                    # Roll for Camp Fire tree
                    add_domicile_building = camp_fire_01
                    add_domicile_building = camp_fire_02
                    random_list = {
                        10 = { add_domicile_building = camp_fire_trailing_musicians }
                        10 = { add_domicile_building = camp_fire_wandering_poets }
                        10 = { add_domicile_building = camp_fire_capering_fools }
                        10 = { add_domicile_building = camp_fire_libations_for_the_lost }
                    }
                }
            }
            
            # --- TIER 3 ROLLS ---
            random_list = {
                10 = {
                    # Roll for Security Perimeter
                    add_domicile_building = camp_perimeter_01
                    add_domicile_building = camp_perimeter_02
                    random_list = {
                        10 = { add_domicile_building = camp_perimeter_good_siting }
                        10 = { add_domicile_building = camp_perimeter_fixed_layout }
                    }
                }
                10 = {
                    # Roll for Basic Proving Grounds
                    add_domicile_building = proving_grounds_01
                    add_domicile_building = proving_grounds_02
                }
            }
        }

        # --- PHASE 3: SPOUSES FORCED RELOCATION ---
        # Changed back to an independent IF block so it doesn't get skipped!
        if = {
            limit = { has_ep3_dlc_trigger = yes }
            
            character:nars_0004 ?= {
                if = {
                    limit = { character:nars_0001 = { is_alive = yes } }
                    character:nars_0001 = { add_courtier = character:nars_0004 }
                }
            }
            character:nars_0005 ?= {
                if = {
                    limit = { character:nars_0002 = { is_alive = yes } }
                    character:nars_0002 = { add_courtier = character:nars_0005 }
                }
            }
            character:nars_0006 ?= {
                if = {
                    limit = { character:nars_0003 = { is_alive = yes } }
                    character:nars_0003 = { add_courtier = character:nars_0006 }
                }
            }
        }
    }    
}
