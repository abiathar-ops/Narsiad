		is_available = yes
		is_at_war = no
	}
	
	effect = {
		custom_tooltip = decision_develop_industry_tt
		show_as_tooltip = {
			add_character_modifier = {
				modifier = industrial_boom_character_modifier
				years = 10
			}
		}
		hidden_effect = {
			add_character_modifier = {
				modifier = industrial_boom_character_modifier
				years = 10
			}
		}
		every_de_jure_county = {
				limit = {
					development_level >= 6
				}
				add_county_modifier = {
					modifier = industrial_boom
					years = 10
				}
		}
		hidden_effect = { #To make the effect screen cleane
			every_de_jure_county = {
					limit = {
						development_level >= 6
					}
					title_province = {
						add_province_modifier = {
							modifier = industrial_boom_province_modifier
							years = 10
						}
					}
				}
		}
		trigger_event = {
			id = narsid_realm.1001
			days = 1
		}	
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_created_develop_industry
			}
			set_global_variable = {
				name = flag_created_develop_industry
				value = scope:founder
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 50
	}
