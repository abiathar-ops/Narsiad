		reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_hostility.dds" #correct
	}

	ai_goal = yes
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
	selection_tooltip = decision_try_for_a_child_tooltip
	desc = decision_try_for_a_child_desc
	confirm_text = decision_try_for_a_child_confirm

	decision_group_type = narsid_minor

	is_shown = {
		is_narsid = yes
		trigger_if = {
			limit = { is_married = no }
			always = no
		}
		trigger_else_if = {
			limit = { is_married = yes }
			sex_opposite_of = primary_spouse
			NOR = {
				has_trait = pregnant
				primary_spouse = { has_trait = pregnant }
			}
			OR = {
				is_ai = no
				AND = {
					is_ai = yes
					primary_spouse = { is_ai = yes }
				}
			}
		}
		trigger_else = {
			always = no
		}
	}

	is_valid = {
		trigger_if = {
			limit = { is_married = no }
			always = no
		}
		trigger_else_if = {
			limit = { is_married = yes }
			stress_level < 1
			is_at_war = no
			NOR = {
				has_trait = pregnant
				primary_spouse = { 	has_trait = pregnant }
			}
		}
		trigger_else = {
			always = no
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
	}

	effect = {
		trigger_event = narsid_racial.0001
		custom_tooltip = decision_try_for_a_child_effect
	}
}
