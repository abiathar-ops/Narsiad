					trigger_event = { id = narsid_racial.1003 }

	}

	ai_will_do = {
		base = 10 # It's nice to do even if you don't have stress
		modifier = {
			add = 100

decision_introduce_bigamy = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
	}
	desc = decision_introduce_bigamy_desc
	selection_tooltip = decision_introduce_bigamy_tooltip
	decision_group_type = narsid_minor

	is_shown = {
		is_narsid = yes
		OR = {
			has_title = title:e_narsokratia
			has_title = title:h_narsokratia
		}
		top_liege = this
		is_capable_adult = yes
		is_landed = yes
	}

	is_valid = {
		is_imprisoned = no
		faith = {
			NOT = { has_doctrine = doctrine_bigamy }
		}
		piety_level >= 3
	}

	cost = {
		piety = 1250
		prestige = 625
	}

	effect = {
		custom_tooltip = decision_narsid_introduce_bigamy_effect_tooltip
		
		# Change the faith's marriage doctrine to allow multiple spouses
		faith = {
			if = {
				limit = { has_doctrine = doctrine_monogamy }
				remove_doctrine = doctrine_monogamy
			}
			if = {
				limit = { has_doctrine = doctrine_concubines }
				remove_doctrine = doctrine_concubines
			}
				if = {
				limit = { has_doctrine = doctrine_polygamy }
				remove_doctrine = doctrine_polygamy
			}
			add_doctrine = doctrine_bigamy
		}
		
		hidden_effect = {
			# trigger_event = { id = narsid_realm.3001 }
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 50
	}
}

decision_establish_aretaic_school = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_golden_age.dds"
	}
	decision_group_type = narsid_major
	ai_check_interval = 0

	desc = decision_establish_aretaic_school_desc
	selection_tooltip = decision_establish_aretaic_school_tooltip

	is_shown = {
		is_narsid = yes        
		OR = {
			has_title = title:k_narsokratia
			has_title = title:e_narsokratia
			has_title = title:h_narsokratia
		}
		NOT = { exists = global_var:established_aretaic_school } # Can only be built once globally
		exists = capital_barony
		capital_province = {
			NOT = { exists = var:aretaic_school }
		}
	}

	is_valid = {
		prestige_level >= 3
		OR = {
			learning >= 15
			martial >= 15
		}
		trigger_if = {
			limit = {
				exists = capital_barony
				capital_province = { NOT = { exists = var:aretaic_school } }
			}
			AND = {
				capital_county = { development_level >= 30 }
				custom_description = {
					text = no_special_building_slot
					capital_province = { has_special_building_slot = no }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_ruler = yes
		is_available_adult = yes
		is_imprisoned = no
	}

	cost = {
		gold = 1000
		prestige = 1000
	}

	effect = {
		save_scope_as = founder

		custom_tooltip = decision_establish_aretaic_school_effect_tooltip

		# Lock it so no one else can take the decision
		set_global_variable = {
			name = established_aretaic_school
			value = yes
		}

		# Add the special building slot and the building itself to the map
		capital_province = {
			set_variable = {
				name = aretaic_school
				value = yes
			}
			add_special_building_slot = university_special_building_slot
			add_special_building = aretaic_school_building
		}

		# Give the founder their lifelong modifier
		add_character_modifier = {
			modifier = aretaic_founder_modifier
			years = -1
		}

		add_martial_lifestyle_xp = 500
		add_learning_lifestyle_xp = 500
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
