decision_establish_aretaic_school = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/ep3_medi_market.dds"
	}
	decision_group_type = realm
	desc = decision_establish_aretaic_school_desc
	selection_tooltip = decision_establish_aretaic_school_tooltip
	sort_order = 850
	
	is_shown = {
		is_selian = yes
		highest_held_title_tier >= tier_kingdom
		NOT = { government_has_flag = government_is_tribal }
		NOT = { government_has_flag = government_is_nomadic }
		NOT = { exists = global_var:established_aretaic_school }
		any_held_title = {
			title_tier = barony
			title_province = { has_special_building_slot = no }
			is_leased_out = no
		}
		is_ai = no
	}

	is_valid = {
		gold >= 1000
		OR = {
			prestige_level >= 4
			learning >= 25
		}
		custom_tooltip = {
			text = establish_aretaic_school_no_barony_tt
			any_held_title = {
				title_tier = barony
				title_province = { has_special_building_slot = no }
				is_leased_out = no
			}
		}
	}

	effect = {
		custom_tooltip = decision_establish_aretaic_school_effect_tooltip
		open_view_data = {
			view = decision_detail
			data = decision:pick_aretaic_school_barony
			player = root
		}
	}
	
	ai_will_do = {
		base = 0
	}
}

pick_aretaic_school_barony = {
	is_shown = {
		is_ai = no
	}
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/ep3_medi_market.dds"
	}
	decision_group_type = realm
	is_invisible = yes
	
	cost = { gold = 1000 }

	widget = {
		controller = create_holy_order
		barony_valid = {
			trigger_if = {
				limit = { exists = this }
				is_barony_valid_for_aretaic_school = {
					HOLDER = scope:ruler
				}
			}
		}
	}

	effect = {
		custom_tooltip = pick_special_building_barony_custom_tt
		set_global_variable = {
			name = established_aretaic_school
			value = yes
		}
		hidden_effect = {
			scope:barony.title_province = {
				set_variable = {
					name = university
					value = yes
				}
				add_special_building = aretaic_school_01
			}
		}
	}
	
	ai_will_do = {
        base = 0
	}
}

decision_establish_brothel = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/ep3_medi_market.dds"
	}
	decision_group_type = realm
	desc = decision_establish_brothel_desc
	selection_tooltip = decision_establish_brothel_tooltip
	sort_order = 850
		
	is_shown = {
		is_selian = yes
		highest_held_title_tier >= tier_kingdom
		OR = {
			exists = global_var:established_aretaic_school
			is_ai = yes

		}
		any_held_title = {
			tier = tier_barony
			is_barony_valid_for_brothel_complex = {
				HOLDER = root
			}
		}
		NOR = { 
			government_has_flag = government_is_nomadic 
			government_has_flag = government_is_tribal
		}
	}

	is_valid = {
		gold >= 750
		OR = {
			prestige_level >= 4
			intrigue >= 25
		}
		custom_tooltip = {
			text = establish_brothel_no_barony_tt
			any_held_county = {
				any_in_de_jure_hierarchy = {
					is_barony_valid_for_brothel_complex = {
						HOLDER = root
					}
				}
			}
		}
		NOR = {
			custom_tooltip = {
				text = establish_brothel_cooldown_tt
				has_character_flag = establish_brothel_cooldown
			}
		}
	}

	effect = {
		custom_tooltip = decision_establish_brothel_effect_tooltip
		open_view_data = {
			view = decision_detail
			data = decision:pick_brothel_barony
			player = root
		}
	}
	
	ai_check_interval = 13
	ai_potential = {
		is_ruler = yes
		any_held_county = {
			any_in_de_jure_hierarchy = {
                is_barony_valid_for_brothel_complex = {
					HOLDER = root
				}
			}
		}
	}
	ai_will_do = {
        base = 100
        modifier = {
			add = -50
			any_held_county = {
				any_in_de_jure_hierarchy = {
					title_province = {
						has_building_or_higher = brothel_complex_01
					}
				}
			}
		}
	}
}
