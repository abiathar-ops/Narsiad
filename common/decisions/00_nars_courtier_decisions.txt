decision_recruit_selian_courtier = {
	ai_check_interval = 0
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_secret_faith.dds" 
	}
	sort_order = 1000
	decision_group_type = courtier

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 60
		hegemony = 60
	}
	
	cost = {
		prestige = major_prestige_value
	}

	is_shown = {
		is_landed_or_landless_administrative = yes
		is_selian = yes
	}

	is_valid = {
		is_at_war = no
		is_imprisoned = no
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_adult = yes
		is_imprisoned = no
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button =  "SELECT_ACTION"
		show_from_start = yes

		item = { # 2 Normal Selians
			value = decision_hire_selian_1
			is_shown = {
				is_ruler = yes
				is_landless_adventurer = no
				is_selian = yes
			}
			is_valid = {
				is_selian = yes
				is_at_war = no
				is_imprisoned = no
				NOR = {
					custom_tooltip = {
						text = recruit_selian_1_cooldown_tt
						has_character_flag = recruit_selian_1_cooldown
					}
				}
			}
			current_description = decision_hire_selian_1_desc
			localization = decision_hire_selian_1
			icon = "gfx/interface/icons/court_position_types/yeke_jarquchi_court_position.dds"
			ai_chance = { value = 100 }
		} 
		item = { # Choice of 2 really good selians
			value = decision_hire_selian_2
			is_shown = {
				exists = primary_title 
				is_landless_adventurer = no
				is_selian = yes
			}
			is_valid = {
				is_at_war = no
				is_imprisoned = no
				highest_held_title_tier >= tier_kingdom
				gold >= medium_treasury_or_gold_value
				NOR = {
					custom_tooltip = {
						text = recruit_selian_2_cooldown_tt
						has_character_flag = recruit_selian_2_cooldown
					}
				}
			}
			current_description = decision_hire_selian_2_desc
			localization = decision_hire_selian_2
			icon = "gfx/interface/icons/court_position_types/champion_court_position.dds"
			ai_chance = {
				value = 100
				if = {
					limit = {
						NOR = {
							highest_held_title_tier > tier_barony
							is_landed_or_landless_administrative = yes
							gold > medium_treasury_or_gold_value
							is_at_war = no
						}
					}
					multiply = 0
				}
			}
		}
	}

	effect = {
		if = {
			limit = { scope:decision_hire_selian_1 = yes }
			custom_tooltip = decision_hire_selian_1_tooltip
		}
		else_if = {
			limit = { scope:decision_hire_selian_2 = yes }
			custom_tooltip = decision_hire_selian_2_tooltip
		}
		hidden_effect = {
			custom_tooltip = decision_recruit_selian_courtier_tooltip
			if = { # 3 men of narses 
				limit = { scope:decision_hire_selian_1 = yes }
				set_variable = {
					name = recruit_selian_1_cooldown
					years = 10
				}
				set_variable = {
					name = number_of_invite_selian_countdown
					value = 4
					days = 1000
					}
				trigger_event = {
					id = nars_courtiers.0001
					days = { 5 15 }
				}
			}
			else_if = { # 2 Sophes
				limit = { scope:decision_hire_selian_2 = yes }
				add_character_flag = {
					flag = recruit_selian_2_cooldown
					years = 5
				}
				trigger_event = {
					id = nars_courtiers.0011
					days = { 7 14 }
				}
			}
		}
	}
	
	ai_potential = {
		is_selian = yes
		is_landed_or_landless_administrative = yes
		modifier = {
			add = 30
			court_size < 10
		}
	}
}
