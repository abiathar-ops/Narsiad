decision_recruit_nars = {
	ai_check_interval = 15
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_akropolis.dds" 
	}
	sort_order = 1000
	decision_group_type = courtier
	
	cost = {
		prestige = major_prestige_value
	}

	is_shown = {
		OR = {
			is_ai = no
			max_number_of_knights > 0
		}
		is_landed_or_landless_administrative = yes
		is_selian = yes
	}

	# is_valid = {
	# 	is_at_war = no
	# 	is_imprisoned = no
	# }

	is_valid_showing_failures_only = {
		is_alive = yes
		not = { exists = involved_activity }
		is_imprisoned = no
		is_at_war = no
		is_adult = yes
		has_contagious_deadly_disease_trigger = no
		custom_tooltip = {
			text = decision_recruit_nars_ongoing
			NOT = {
				any_scheme = {
					type = recruit_nars_scheme
				}
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = decision_recruit_nars_start_tt
			start_scheme = {
				type = recruit_nars_scheme
				target_character = root
			}
			close_view = {
				view = decisions
				player = root
			}
			random_scheme = {
				type = recruit_nars_scheme
				save_scope_as = scheme
			}
			save_scope_as = owner
		}
	}
	
	ai_potential = {
		base = 0
		is_selian = yes
		is_landed_or_landless_administrative = yes
		modifier = {
			add = 30
			court_size < 10
		}
	}
}
