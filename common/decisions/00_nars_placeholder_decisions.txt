#Adds trait that gives events that boost MAA
decision_start_cultivate_hoplites = {
	picture = {
 		reference = "gfx/interface/illustrations/decisions/mpo_decision_confederation.dds"
 	}
 	desc = decision_mercenary_spawn_desc
 	decision_group_type = nikean
 	selection_tooltip = decision_mercenary_spawn_tooltip

 	is_shown = {
 		is_nikean = yes
 		top_liege = this
 		is_capable_adult = yes

 		NOT = {
 			is_target_in_global_variable_list = {
 				name = unavailable_unique_decisions
 				target = flag:flag_mend_narsil_decision
 			}
 		}
 	}

 	is_valid = {
		is_playable = yes
 		is_nikean = yes
 		is_imprisoned = no
 		is_at_war = no
 	}

 	is_valid_showing_failures_only = {
 		top_liege = this
 		is_available_adult = yes
 		is_at_war = no
 	}

 	effect = {
 	}
 	ai_potential = { always = yes }

 	ai_will_do = {
 		base = 10
		if = {
			limit = { gold >= 500 }
			modifier = {
				factor = 5
			}
		}
 	}
}
