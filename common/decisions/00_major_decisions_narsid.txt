decision_h_narsokratia = { #Narsil and e_narsokratia
	picture = {
		reference = "gfx/interface/illustrations/decisions/narsokratia.dds"
	}
	desc = decision_h_narsokratia_desc
	selection_tooltip = decision_h_narsokratia_tooltip
	decision_group_type = narsid_major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 120
		hegemony = 0
	}

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		is_narsid = yes
		has_title = title:e_narsokratia # Only the Narsid Empire can see the Hegemony decision
		NOT = { has_title = title:h_narsokratia }
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_h_narsokratia_decision
			}
		}
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_republic
			government_has_flag = government_is_clan
			government_allows = administrative
		}
	}

	is_valid = {
		prestige_level >= max_prestige_level
		top_liege = this
		has_title = title:e_narsokratia # Required to get H_Narsokratia
		OR = {
			custom_description = {
				text = found_hegemony_decision_eight_kingdoms_held
				any_held_title = {
					count >= 7
					title_tier = kingdom
				}
			}
			sub_realm_size >= 200
		}
		custom_description = {
			text = require_narsil_artifact_tooltip
			any_character_artifact = {
				artifact_template = narsil_template
			}
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
		OR = {
			religion = { is_in_family = rf_narsid }
			culture = { has_cultural_pillar = heritage_nars }
		}
	}
	
	cost = {
		gold = {
			value = 2500
			if = {
				limit = {
					OR = {
						government_has_flag = government_is_nomadic
						has_treasury = yes
					}
				}
				multiply = 0
			}
		}
		treasury = {
			value = 2500
			if = {
				limit = {
					has_treasury = no
				}
				multiply = 0
			}
		}
		piety = {
			value = 1250
			if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				multiply = 0
			}
		}
	}
	
	effect = {
		custom_tooltip = decision_h_narsokratia_effect_tooltip
		create_h_narsokratia_effect = yes
		change_government = administrative_government
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_h_narsokratia_decision
			}
			set_global_variable = {
				name = flag_h_narsokratia_decision
				value = root
			}
		}
	}
		
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

decision_e_narsokratia = { #foundry and k_narsokratia
	picture = {
		reference = "gfx/interface/illustrations/decisions/narsokratia.dds"
	}
	decision_group_type = narsid_major
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 0
		hegemony = 0
	}
	desc = decision_e_narsokratia_desc

	is_shown = {
		is_narsid = yes
		is_landed_or_landless_administrative = yes
		is_confederation_member = no
		NOT = { has_title = title:e_narsokratia }
		NOT = { has_title = title:h_narsokratia }
		OR = {
			has_title = title:k_narsokratia # Narsid Kingdoms can see it
			highest_held_title_tier = tier_empire # Generic Empires can see it
		}
	}

	is_valid = {
		prestige_level >= 4
		top_liege = this
		OR = {
			has_title = title:k_narsokratia # Required if upgrading from Kingdom
			highest_held_title_tier = tier_empire # Generic Empires bypass the K requirement
		}
		trigger_if = {
			limit = {
				government_has_flag = government_is_nomadic
			}
			OR = {
				sub_realm_size >= 60
				AND = {
					custom_description = {
						text = found_e_narsokratia_decision_three_kingdoms_held
						any_held_title = {
							count >= 2
							title_tier = kingdom
						}
					}
					sub_realm_size >= 40
				}
			}
			OR = {
				custom_description = {
					text = found_e_narsokratia_decision_three_kingdoms_held
					any_held_title = {
						count >= 2
						title_tier = kingdom
					}
				}
				sub_realm_size >= 40
			}
		}
		trigger_if = {
			limit = {
				highest_held_title_tier = tier_empire
			}
		}
		trigger_else = {
			OR = {
				custom_description = {
					text = found_e_narsokratia_decision_three_kingdoms_held
					any_held_title = {
						count >= 3
						title_tier = kingdom
					}
				}
				sub_realm_size >= 80
			}
		}
	}

	is_valid_showing_failures_only = {
		top_liege = this
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = {
			value = 1250
			if = {
				limit = {
					OR = {
						government_has_flag = government_is_nomadic
						has_treasury = yes
					}
				}
				multiply = 0
			}
		}
		treasury = {
			value = 1250
			if = {
				limit = {
					has_treasury = no
				}
				multiply = 0
			}
		}
		piety = {
			value = 625
			if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				multiply = 0
			}
		}
	}

	effect = {
		change_government = administrative_government
		custom_tooltip = decision_e_narsokratia_effect_tooltip
		create_narsid_empire_effect = yes
		if = {
			limit = {
				is_ai = no
				NOT = { exists = global_var:found_e_narsokratia_decision  }
			}
			set_global_variable = {
				name = found_e_narsokratia_decision
				value = root
			}
			set_global_variable = {
				name = found_e_narsokratia_decision_empire
				value = root.primary_title
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
