decision_h_narsokratia = { #Narsil and e_narsokratia
	picture = {
		reference = "gfx/interface/illustrations/decisions/narsokratia.dds"
	}
	desc = decision_h_narsokratia_desc
	selection_tooltip = decision_h_narsokratia_tooltip
	decision_group_type = narsid

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 120
		hegemony = 0
	}
	
	sort_order = 1000

	is_shown = {
		is_ruler = yes
		is_narsid = yes
		has_title = title:e_narsokratia # Only the Narsid Empire can see the Hegemony decision
		top_liege = this
		NOT = { has_title = title:h_narsokratia }
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_h_narsokratia_decision
			}
		}
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_republic
			government_has_flag = government_is_clan
			government_allows = administrative
		}
	}

	is_valid = {
		has_title = title:e_narsokratia # Required to get H_Narsokratia
		prestige_level >= max_prestige_level
		OR = {
			custom_description = {
				text = found_hegemony_decision_eight_kingdoms_held
				any_held_title = {
					count >= 7
					title_tier = kingdom
				}
			}
			sub_realm_size >= 200
		}
		custom_description = {
			text = require_narsil_artifact_tooltip
			any_character_artifact = {
				artifact_template = narsil_template
			}
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
		OR = {
			religion = { is_in_family = rf_narsid }
			culture = { has_cultural_pillar = heritage_nars }
		}
	}
	
	cost = {
		gold = {
			value = 2500
			if = {
				limit = {
					OR = {
						government_has_flag = government_is_nomadic
						has_treasury = yes
					}
				}
				multiply = 0
			}
		}
		treasury = {
			value = 2500
			if = {
				limit = {
					has_treasury = no
				}
				multiply = 0
			}
		}
		piety = {
			value = 1250
			if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				multiply = 0
			}
		}
	}
	
	effect = {
		custom_tooltip = decision_h_narsokratia_effect_tooltip
		create_h_narsokratia_effect = yes
		add_trait = nars_trait
		change_government = administrative_government
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_h_narsokratia_decision
			}
			set_global_variable = {
				name = flag_h_narsokratia_decision
				value = root
			}
			local_assimilate_narsid_traditions_effect = {
				OLD_CULTURE = root.culture 
				NEW_CULTURE = root.culture
				OLD_FAITH = root.faith
				NEW_FAITH = root.faith
				CONVERTER = root
			}
		}
	}
		
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
