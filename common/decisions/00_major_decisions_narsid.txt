decision_e_narsokratia = { #foundry and k_narsokratia
	picture = {
		reference = "gfx/interface/illustrations/decisions/narsokratia.dds"
	}
	decision_group_type = narsid_major
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 0
		hegemony = 0
	}
	desc = decision_e_narsokratia_desc

	is_shown = {
		is_narsid = yes
		highest_held_title_tier = tier_kingdom
		OR = {
			top_liege = this
			any_held_title = {
				count >= 3
				title_tier = kingdom
			}
		}
		is_landed_or_landless_administrative = yes
		is_confederation_member = no
	}

	is_valid = {
		prestige_level >= 4
		top_liege = this
		trigger_if = {
			limit = {
				government_has_flag = government_is_nomadic
			}
			OR = {
				sub_realm_size >= 60
				AND = {
					custom_description = {
						text = found_e_narsokratia_decision_three_kingdoms_held
						any_held_title = {
							count >= 2
							title_tier = kingdom
						}
					}
					sub_realm_size >= 40
				}
			}
			OR = {
				custom_description = {
					text = found_e_narsokratia_decision_three_kingdoms_held
					any_held_title = {
						count >= 2
						title_tier = kingdom
					}
				}
				sub_realm_size >= 40
			}
		}
		trigger_else = {
			OR = {
				custom_description = {
					text = found_empire_decision_three_kingdoms_held
					any_held_title = {
						count >= 3
						title_tier = kingdom
					}
				}
				sub_realm_size >= 80
			}
		}
	}

	is_valid_showing_failures_only = {
		top_liege = this
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = {
			value = 1250
			if = {
				limit = {
					OR = {
						government_has_flag = government_is_nomadic
						has_treasury = yes
					}
				}
				multiply = 0
			}
		}
		treasury = {
			value = 1250
			if = {
				limit = {
					has_treasury = no
				}
				multiply = 0
			}
		}
		piety = {
			value = 625
			if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				multiply = 0
			}
		}
	}

	effect = {
		change_government = meritocratic_government
		custom_tooltip = decision_e_narsokratia_effect_tooltip
		create_narsid_empire_effect = yess
		if = {
			limit = {
				is_ai = no
				NOT = { exists = global_var:found_e_narsokratia_decision  }
			}
			set_global_variable = {
				name = found_e_narsokratia_decision
				value = root
			}
			set_global_variable = {
				name = found_e_narsokratia_decision_empire
				value = root.primary_title
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
