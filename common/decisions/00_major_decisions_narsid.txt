	picture = {
			reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
		}
		desc = found_h_narsokratia_decision_desc
		selection_tooltip = found_h_narsokratia_decision_tooltip
		decision_group_type = narsid_major

		ai_check_interval_by_tier = {
			barony = 0
			county = 0
			duchy = 0
			kingdom = 0
			empire = 120
			hegemony = 0
		}

		is_shown = {
			is_ruler = yes
			is_playable_character = yes
			is_narsid = yes
			NOT = {  #Can only do it once.
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:flag_h_narsokratia_decision
				}
			}
			OR = {
				government_has_flag = government_is_feudal
				government_has_flag = government_is_republic
				government_has_flag = government_is_clan
				government_allows = administrative
			}
		}

		is_valid = {
			prestige_level >= max_prestige_level
			has_title = title:e_narsokratia
			custom_description = {
				text = found_hegemony_decision_eight_kingdoms_held
				any_held_title = {
				count >= 7
				title_tier = kingdom
				}
			}
			culture != capital_province.culture
		}

		is_valid_showing_failures_only = {
			is_landed = yes
			is_capable_adult = yes
			is_imprisoned = no
			top_liege = this
			OR = {
				religion = { is_in_family = rf_narsid }
				culture = { has_cultural_pillar = heritage_nars }
			}
		}

		effect = {
			custom_tooltip = found_h_narsokratia_decision_effect_tooltip
			show_as_tooltip = {
				create_h_narsokratia_scripted_effect = yes
			}
			if = {
				limit = {
					has_ep3_dlc_trigger = yes
				}
				trigger_event = narsid_realm.2001
			}
			else = {
				trigger_event = narsid_realm.2001
			}
			hidden_effect = {
				add_to_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:flag_h_narsokratia_decision
				}
				set_global_variable = {
					name = flag_h_narsokratia_decision
					value = root
				}
			}
			local_assimilate_narsid_traditions_effect = {
				CONVERTER = root
				OLD_CULTURE = root.culture
				OLD_FAITH = root.faith
				NEW_CULTURE = capital_province.culture
				NEW_FAITH = capital_province.faith
			}
		}
		
		ai_potential = {
			always = yes
		}

		ai_will_do = {
			base = 100
		}
}

decision_e_narsokratia = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	decision_group_type = narsid_major
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 0
		hegemony = 0
	}
	desc = found_e_narsokratia_decision_desc

	is_shown = {
		is_narsid = yes
		highest_held_title_tier = tier_kingdom
		OR = {
			top_liege = this
			any_held_title = {
				count >= 3
				title_tier = kingdom
			}
		}
		is_landed_or_landless_administrative = yes
		is_confederation_member = no
	}

	is_valid = {
		prestige_level >= 4
		trigger_if = {
			limit = {
				government_has_flag = government_is_nomadic
			}
			OR = {
				sub_realm_size >= 60
				AND = {
					custom_description = {
						text = found_e_narsokratia_decision_three_kingdoms_held
						any_held_title = {
							count >= 2
							title_tier = kingdom
						}
					}
					sub_realm_size >= 40
				}
			}
		}
		trigger_else = {
			OR = {
				sub_realm_size >= 120
				AND = {
					custom_description = {
						text = found_empire_decision_three_kingdoms_held
						any_held_title = {
							count >= 3
							title_tier = kingdom
						}
					}
					sub_realm_size >= 80
				}
			}
		}

		rule_title_creation_imperial_power_projection_title_creation_trigger = yes
	}

	is_valid_showing_failures_only = {
		top_liege = this
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = {
			value = 1200
			if = {
				limit = {
					OR = {
						government_has_flag = government_is_nomadic
						has_treasury = yes
					}
				}
				multiply = 0
			}
		}
		treasury = {
			value = 1200
			if = {
				limit = {
					has_treasury = no
				}
				multiply = 0
			}
		}
		prestige = 2000
		piety = {
			value = 600
			if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				multiply = 0
			}
		}
	}

	effect = {
		if = {
			limit = {
				NOT = { government_has_flag = government_is_nomadic }
			}
			create_custom_empire_effect = yes
			gain_heroic_legend_seed_tooltip_effect = yes
			hidden_effect = {
				legend_seed_new_title_effect = yes
			}
		}
		else = {
			create_custom_empire_effect = yes
		}
		if = {
			limit = {
				is_ai = no
				NOT = { exists = global_var:found_e_narsokratia_decision  }
			}
			set_global_variable = {
				name = found_e_narsokratia_decision
				value = root
			}
			set_global_variable = {
				name = found_e_narsokratia_decision_empire
				value = root.primary_title
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
