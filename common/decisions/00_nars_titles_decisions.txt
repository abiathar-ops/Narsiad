decision_pledge_restoration = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/narsokratia.dds"
    }
    desc = decision_pledge_restoration_desc
    selection_tooltip = decision_pledge_restoration_tooltip
    decision_group_type = restorationist
    sort_order = 1000

    is_shown = {
        is_ruler = yes
        is_independent_ruler = yes
        is_ethnically_minoan = yes
        is_minoan = yes
        highest_held_title_tier >= tier_kingdom
        dynasty = {
            NOT = { has_dynasty_modifier = pledge_restoration_dynasty_modifier }
        }
    }

    is_valid = {
        # Keep your mirrored Conqueror requirements here as they were
        highest_held_title_tier >= tier_kingdom
        prestige_level >= 5
        custom_tooltip = { text = become_conqueror_decision_war_participation any_memory = { count >= 20 OR = { has_memory_type = offensive_war has_memory_type = defensive_war has_memory_type = joined_allys_war } } }
        custom_tooltip = { text = become_conqueror_decision_war_participation_kingdom any_memory = { count >= 5 OR = { has_memory_type = offensive_war has_memory_type = defensive_war has_memory_type = joined_allys_war } any_memory_participant = { primary_title.tier >= tier_kingdom } } }
        custom_tooltip = { text = nars_conqueror_decision_enough_troops current_military_strength >= 3500 }
        calc_true_if = { amount >= 1 custom_tooltip = { text = become_conqueror_decision_enough_high_level_knights any_knight = { count >= 6 prowess >= 18 } } custom_tooltip = { text = become_conqueror_decision_enough_mediocre_knights any_knight = { count >= 12 prowess >= 15 } } }
    }

    is_valid_showing_failures_only = {
        is_landed = yes
        is_capable_adult = yes
        is_imprisoned = no
        OR = { is_orphic = yes is_minoan = yes }
    }
    
    cost = { prestige = 5000 }
    
    effect = {
        # UI Tooltips explaining what happens upon Victory
        custom_tooltip = nars_conqueror_defensive_war_tt
        custom_tooltip = gain_nars_conqueror_tt
		custom_tooltip = decision_pledge_restoration_tt
        custom_tooltip = nars_conqueror_decision_troops
        if = { limit = { has_dlc_feature = legends } custom_tooltip = become_conqueror_decision_legend_seed }

        add_character_flag = flag_decision_pledge_restoration

		every_neighboring_top_liege_realm_owner = {
            limit = {
                is_independent_ruler = yes
                NOT = { has_alliance = root }
                NOT = { has_truce = root }
            }
            show_as_tooltip = {
                custom_description_no_bullet = {
                    text = minoan_coalition_member_name_tt
                }
            }
        }

		hidden_effect = {
			# 1. Build a list of all strictly land-bordering independent neighbors
			every_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					is_independent_ruler = yes
					NOT = { has_alliance = root }
					NOT = { has_truce = root }
				}
				add_to_list = minoan_coalition_members
			}

			if = {
				limit = {
					any_in_list = { list = minoan_coalition_members }
				}
				
				# 2. Pick the strongest neighbor to lead the Coalition
				ordered_in_list = {
					list = minoan_coalition_members
					order_by = current_military_strength
					save_scope_as = coalition_war_leader
				}

				# 3. The War Leader declares the war
				scope:coalition_war_leader = {
					start_war = {
						target = root
						target_title = root.primary_title
						cb = minoan_defiance_cb
					}
				}

				# 4. Save the war, then forcefully drag the rest of the list into it
				root = {
					random_character_war = {
						limit = { primary_attacker = scope:coalition_war_leader }
						
						every_in_list = {
							list = minoan_coalition_members
							limit = {
								NOT = { this = scope:coalition_war_leader }
							}
							# 'prev' scopes back to the war, adding the list character as an attacker
							prev = {
								add_attacker = prev
							}
						}
					}
				}
			}
		}
    }
        
    ai_potential = { always = yes }
    ai_will_do = { base = 100 }
}
