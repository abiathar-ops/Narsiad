decision_recruit_nars = {
	ai_check_interval = 15
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_akropolis.dds" 
	}
	sort_order = 1000
	decision_group_type = courtier
	
	cost = {
		prestige = major_prestige_value
	}

	is_shown = {
		OR = {
			is_ai = no
			max_number_of_knights > 0
		}
		is_landed_or_landless_administrative = yes
		is_selian = yes
	}

	is_valid_showing_failures_only = {
		is_alive = yes
		not = { exists = involved_activity }
		is_imprisoned = no
		is_at_war = no
		is_adult = yes
		has_contagious_deadly_disease_trigger = no
		custom_tooltip = {
			text = decision_recruit_nars_ongoing
			NOT = {
				any_scheme = {
					type = recruit_nars_scheme
				}
			}
		}
		OR = {
			is_ai = no
			max_number_of_knights > 0
		}
	}

	effect = {
		custom_tooltip = {
			text = decision_recruit_nars_start_tt
			start_scheme = {
				type = recruit_nars_scheme
				target_character = root
			}
			close_view = {
				view = decisions
				player = root
			}
			random_scheme = {
				type = recruit_nars_scheme
				save_scope_as = scheme
			}
			save_scope_as = owner
		}
	}
	
	ai_potential = {
		base = 0
		is_selian = yes
		is_landed_or_landless_administrative = yes
		modifier = {
			add = 30
			court_size < 10
		}
	}
}

decision_mend_narsil = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_domination.dds"
	}
	
	desc = decision_mend_narsil_desc
	decision_group_type = major
	selection_tooltip = decision_mend_narsil_tooltip

	is_shown = {
		is_selian = yes
		highest_held_title_tier >= tier_kingdom
		top_liege = this
		is_capable_adult = yes
		is_landed = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mend_narsil_decision
			}
		}
	}

	is_valid = {
		custom_tooltip = {
			text = decision_mend_narsil_req_tt
			any_character_artifact = { has_variable = can_be_used_for_narsil }
		}
	}

	is_valid_showing_failures_only = {
		top_liege = this
		is_available_adult = yes
		is_imprisoned = no
		is_at_war = no
		custom_tooltip = {
			text = decision_mend_narsil_ongoing
			NOT = {
				any_scheme = {
					type = mend_narsil_scheme
				}
			}
		}
	}
	
	sort_order = 900

	effect = {
		custom_tooltip = {
			text = decision_mend_narsil_effect_tooltip
			if = {
				limit = {
					NOT = { any_owned_story = { story_type = story_mend_narsil } }
				}
				create_story = story_mend_narsil
			}

			start_scheme = {
				type = mend_narsil_scheme
				target_character = root
			}

			close_view = {
				view = decisions
				player = root
			}
		}
	}

	ai_check_interval = 0
}
