decision_develop_industry = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/innovation.dds"
	}
	ai_goal = yes
	cost = {
		gold = 625
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 130
		duchy = 130
		kingdom = 130
		empire = 130
		hegemony = 130
	}
	
	sort_order = 600

	decision_group_type = nikean
	
	is_shown = {
		is_nikean = yes
		is_ruler = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_created_develop_industry
			}
		}
		is_landed = yes
		OR = {
			has_title = title:k_narsokratia
			has_title = title:e_narsokratia
			has_title = title:h_narsokratia
		}
	}

	is_valid = {
		is_at_war = no
		is_imprisoned = no
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_available = yes
		is_at_war = no
	}
	
	effect = {
		custom_tooltip = decision_develop_industry_tt
		show_as_tooltip = {
			add_character_modifier = {
				modifier = industrial_boom_character_modifier
				years = 10
			}
		}
		hidden_effect = {
			add_character_modifier = {
				modifier = industrial_boom_character_modifier
				years = 10
			}
		}
		every_de_jure_county = {
			limit = { development_level >= 6 }
			add_county_modifier = {
				modifier = industrial_boom
				years = 10
			}
		}
		hidden_effect = {
			every_de_jure_county = {
				limit = { development_level >= 6 }
				title_province = {
					add_province_modifier = {
						modifier = industrial_boom_province_modifier
						years = 10
					}
				}
			}
		}
		trigger_event = {
			id = nikean_realm.1001
			days = 1
		}	
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_created_develop_industry
			}
			set_global_variable = {
				name = flag_created_develop_industry
				value = scope:founder
			}
		}
	}
	
	ai_potential = { always = yes }

	ai_will_do = {
		base = 50
	}
}
