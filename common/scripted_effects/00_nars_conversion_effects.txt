#  TIERED VASSAL CONVERSION EFFECTS
#  Now include culture conversion alongside faith conversion.
# ─────────────────────────────────────────────────────────────────

# Kingdom tier: ~30% chance per AI vassal converts faith AND culture.
nikean_kingdom_vassal_conversion_effect = {
	every_vassal = {
		limit = { is_ai = yes }
		random = {
			chance = 30
			set_character_faith = root.faith
		}
	}
	every_vassal = {
		limit = { is_ai = yes }
		random = {
			chance = 30
			set_culture = root.culture
		}
	}
}

# Empire tier: ~60% faith + culture conversion + ALL vassals get administrative government.
nikean_empire_vassal_conversion_effect = {
	every_vassal = {
		limit = { is_ai = yes }
		random = {
			chance = 60
			set_character_faith = root.faith
		}
	}
	every_vassal = {
		limit = { is_ai = yes }
		random = {
			chance = 60
			set_culture = root.culture
		}
	}
	# Government: all vassals flipped to administrative
	every_vassal = {
		limit = {
			NOT = { government_has_flag = government_is_administrative }
		}
		change_government = administrative_government
	}
}

# Hegemony tier: FULL government conversion + direct vassals' faith AND culture converted.
nikean_hegemony_vassal_conversion_effect = {
	# Government: every vassal in the realm
	every_vassal = {
		limit = {
			NOT = { government_has_flag = government_is_administrative }
		}
		change_government = administrative_government
	}
	# Faith + culture: direct vassals only (liege = root)
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		set_character_faith = root.faith
	}
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		set_culture = root.culture
	}
}

# Custom script effect to instantly transition Stratō into a Byzantine Estate Owner
strato_procure_byzantine_estate_effect = {
	# --- RIG THE TARGETS ---
	# The Emperor becomes the recipient, Stratō becomes the actor
	title:e_byzantium.holder = { save_scope_as = recipient }
	character:nars_0003 = { save_scope_as = actor }

	# --- STRATŌ TRANSITION ---
	scope:actor = {
		hidden_effect = {
			# Liquidate the landless mercenary camp
			laamp_clear_domicile_buildings_liquidator_effect = yes
		}
		
		# Prevent his courtiers/assassins from scattering during the transition
		every_courtier = { add_to_list = courtiers_list }
		
		# Destroy the Landless Adventurer Camp title
		destroy_laamp_effect = { ADVENTURER = scope:actor }
		
		hidden_effect = {
			# Force Stratō into the Byzantine Emperor's court temporarily
			scope:recipient = {
				add_courtier = scope:actor
				return_to_court = yes
			}
		}
		
		# Grant Stratō his Noble Family and Estate
		create_noble_family_effect = { GOVERNMENT_GIVER = scope:recipient }
		
		hidden_effect = {
			# Return his loyal Narsid courtiers to his new estate
			every_in_list = {
				list = courtiers_list
				limit = {
					NOT = { liege_or_court_owner = scope:actor }
				}
				scope:actor = { add_courtier = prev }
			}
			# Boot any random lowborns the game auto-generated during the transition
			every_courtier = {
				limit = {
					is_lowborn = yes
					NOR = {
						is_in_list = courtiers_list
						is_close_or_extended_family_of = scope:actor
						is_consort_of = scope:actor
						has_important_relationship_trigger = { CHARACTER = scope:actor }
					}
				}
				silent_disappearance_effect = yes
			}
		}
		
		# Swear Fealty to the Byzantine Emperor
		create_title_and_vassal_change = {
			type = swear_fealty
			add_claim_on_loss = no
			save_scope_as = title_change
		}
		change_liege = {
			liege = scope:recipient
			change = scope:title_change
		}
		resolve_title_and_vassal_change = scope:title_change
		
		# Force the physical Estate Domicile to spawn inside Constantinople
		hidden_effect = {
			domicile ?= {
				move_domicile = title:c_byzantion.province
			}
		}
	}
}
