nikean_vassal_conversion_effect = {
	# Government: every vassal in the realm
	every_vassal = {
		change_government = administrative_government
	}
	# Faith + culture: direct vassals only (liege = root)
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		random = {
			chance = 65
			set_character_faith = faith:nars_religion
		}
	}
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		random = {
			chance = 65
			set_culture = culture:nars_nikean
		}
	}
}

nikean_diplomacy_entreaty_effect = {
    custom_tooltip = nikean_diplomacy.tt
    
    hidden_effect = {
        # Dynamically identify the other two siblings based on the root character's modifier
        if = {
            limit = { has_character_modifier = eleios_modifier }
            father = {
                random_child = { limit = { has_character_modifier = ilione_modifier } save_scope_as = target_1 }
                random_child = { limit = { has_character_modifier = strato_modifier } save_scope_as = target_2 }
            }
        }
        else_if = {
            limit = { has_character_modifier = ilione_modifier }
            father = {
                random_child = { limit = { has_character_modifier = eleios_modifier } save_scope_as = target_1 }
                random_child = { limit = { has_character_modifier = strato_modifier } save_scope_as = target_2 }
            }
        }
        else_if = {
            limit = { has_character_modifier = strato_modifier }
            father = {
                random_child = { limit = { has_character_modifier = eleios_modifier } save_scope_as = target_1 }
                random_child = { limit = { has_character_modifier = ilione_modifier } save_scope_as = target_2 }
            }
        }
        
        random_list = {
            20 = {
                modifier = {
                    add = 10
                    scope:target_1 = { opinion: { target = root value >= 50 } }
                    scope:target_2 = { opinion: { target = root value >= 50 } }
                }
                
                send_interface_toast = {
                    title = nikean_diplomacy_both_join_toast_t
                    left_icon = scope:target_1
                    right_icon = scope:target_2
                    custom_tooltip = nikean_diplomacy_both_join_toast_tt
                }
                
                scope:target_1 = {
                    hidden_effect = { laamp_clear_domicile_buildings_liquidator_effect = yes }
                    every_courtier = { add_to_list = target_1_followers_list }
                    destroy_laamp_effect = { ADVENTURER = this }
                    remove_trait = adventurer
                    
                    root = { 
                        add_courtier = scope:target_1 
                        return_to_court = yes 
                    }
                    
                    hidden_effect = {
                        every_in_list = {
                            list = target_1_followers_list
                            limit = { NOT = { liege_or_court_owner = root } }
                            root = { add_courtier = prev }
                        }
                    }
                }
                
                scope:target_2 = {
                    hidden_effect = { laamp_clear_domicile_buildings_liquidator_effect = yes }
                    every_courtier = { add_to_list = target_2_followers_list }
                    destroy_laamp_effect = { ADVENTURER = this }
                    remove_trait = adventurer
                    
                    root = { 
                        add_courtier = scope:target_2 
                        return_to_court = yes 
                    }
                    
                    hidden_effect = {
                        every_in_list = {
                            list = target_2_followers_list
                            limit = { NOT = { liege_or_court_owner = root } }
                            root = { add_courtier = prev }
                        }
                    }
                }
            }
            25 = {
                send_interface_toast = {
                    title = nikean_diplomacy_target_1_joins_toast_t
                    left_icon = scope:target_1
                    right_icon = scope:target_2
                    custom_tooltip = nikean_diplomacy_target_1_joins_toast_tt
                }
                scope:target_1 = {
                    hidden_effect = { laamp_clear_domicile_buildings_liquidator_effect = yes }
                    every_courtier = { add_to_list = target_1_followers_list }
                    destroy_laamp_effect = { ADVENTURER = this }
                    remove_trait = adventurer
                    
                    root = { 
                        add_courtier = scope:target_1 
                        return_to_court = yes 
                    }
                    
                    hidden_effect = {
                        every_in_list = {
                            list = target_1_followers_list
                            limit = { NOT = { liege_or_court_owner = root } }
                            root = { add_courtier = prev }
                        }
                    }
                }
                scope:target_2 = {
                    add_opinion = {
                        target = root
                        modifier = disapproving_opinion
                    }
                    set_relation_rival = {
                        reason = rival_claimant_fight
                        target = root
                    } 
                }
            }
            25 = {
                send_interface_toast = {
                    title = nikean_diplomacy_target_2_joins_toast_t
                    left_icon = scope:target_2
                    right_icon = scope:target_1
                    custom_tooltip = nikean_diplomacy_target_2_joins_toast_tt
                }

                scope:target_1 = {
                    add_opinion = {
                        target = root
                        modifier = disapproving_opinion
                    }
                    set_relation_rival = {
                        reason = rival_claimant_fight
                        target = root
                    } 
                }
                scope:target_2 = {
                    hidden_effect = { laamp_clear_domicile_buildings_liquidator_effect = yes }
                    every_courtier = { add_to_list = target_2_followers_list }
                    destroy_laamp_effect = { ADVENTURER = this }
                    remove_trait = adventurer
                    
                    root = { 
                        add_courtier = scope:target_2 
                        return_to_court = yes 
                    }
                    
                    hidden_effect = {
                        every_in_list = {
                            list = target_2_followers_list
                            limit = { NOT = { liege_or_court_owner = root } }
                            root = { add_courtier = prev }
                        }
                    }
                }
            }
            30 = {
                send_interface_toast = {
                    title = nikean_diplomacy_neither_join_toast_t
                    left_icon = scope:target_1
                    right_icon = scope:target_2
                    custom_tooltip = nikean_diplomacy_neither_join_toast_tt
                }
                scope:target_1 = {
                    add_opinion = {
                        target = root
                        modifier = disapproving_opinion
                        set_relation_rival = {
                            reason = rival_claimant_fight
                            target = root
                        } 
                    }
                }
                scope:target_2 = {
                    add_opinion = {
                        target = root
                        modifier = disapproving_opinion
                        set_relation_rival = {
                            reason = rival_claimant_fight
                            target = root
                        } 
                    }
                }
            }
        }
    }
