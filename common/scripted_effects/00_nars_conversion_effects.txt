selian_vassal_conversion_effect = {
	# Government: every vassal in the realm
	every_vassal = {
		change_government = administrative_government
	}
	# Faith + culture: direct vassals only (liege = root)
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		random = {
			chance = 65
			set_character_faith = faith:nars_religion
		}
	}
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		random = {
			chance = 65
			set_culture = culture:selian
		}
	}
}

selian_settle_county_effect = {
    # Save the character triggering the effect so we can reference them safely
    save_scope_as = selian_ruler
    
    every_sub_realm_county = {
        limit = {
            # Check if the county belongs to a Duchy held by the ruler, 
            # OR if it belongs to the de jure Duchy of the ruler's capital.
            duchy = {
                OR = {
                    holder = scope:selian_ruler
                    this = scope:selian_ruler.capital_county.duchy
                }
            }
            
            # Only target the county if it actually needs a conversion
            OR = {
                NOT = { faith = faith:nars_religion }
                NOT = { culture = culture:selian }
            }
        }
        
        # Convert Faith if needed
        if = {
            limit = {
                NOT = { faith = faith:nars_religion }
            }
            set_county_faith = faith:nars_religion
        }
        
        # Convert Culture if needed
        if = {
            limit = {
                NOT = { culture = culture:selian }
            }
            set_county_culture = culture:selian
        }
        
        # Apply the control loss once
        change_county_control = major_county_control_loss
    }
}

        if = {
            limit = { has_character_modifier = eleios_modifier }
            father = {
                random_child = { limit = { has_character_modifier = ilione_modifier } save_scope_as = target_1 }
                random_child = { limit = { has_character_modifier = strato_modifier } save_scope_as = target_2 }
