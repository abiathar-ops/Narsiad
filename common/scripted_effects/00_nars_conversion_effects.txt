nikean_hegemony_vassal_conversion_effect = {
	# Government: every vassal in the realm
	every_vassal = {
		limit = {
			NOT = { government_has_flag = government_is_administrative }
		}
		change_government = administrative_government
	}
	# Faith + culture: direct vassals only (liege = root)
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		set_character_faith = root.faith
	}
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		set_culture = root.culture
	}
}

bookmark_strato_administrative_event_effect = {
    # --- RIG THE TARGETS ---
    title:e_byzantium.holder = { save_scope_as = recipient }
    character:nars_0003 = { save_scope_as = actor }

    # --- STRATŌ TRANSITION ---
    scope:actor = {
        set_house = house:house_nars_1001

        # 1. SAVE ALL FOLLOWERS BEFORE LIQUIDATION
        every_courtier = { add_to_list = strato_followers }
        every_prisoner = { add_to_list = strato_prisoners }

        hidden_effect = {
            # Liquidate the landless mercenary camp
            laamp_clear_domicile_buildings_liquidator_effect = yes
        }
        
        # 2. DESTROY CAMP
        destroy_laamp_effect = { ADVENTURER = scope:actor }
        
        hidden_effect = {
            # 3. FORCE STRATŌ INTO THE BYZANTINE EMPEROR'S COURT TEMPORARILY
            scope:recipient = {
                add_courtier = scope:actor
                return_to_court = yes
            }
        }
        
        # 4. GRANT STRATŌ HIS NOBLE FAMILY AND ESTATE
        create_noble_family_effect = { GOVERNMENT_GIVER = scope:recipient }
        
        hidden_effect = {
            # 5. RESTORE HIS ENTIRE CAMP OF FOLLOWERS TO HIS NEW ESTATE
            every_in_list = {
                list = strato_followers
                limit = { NOT = { liege_or_court_owner = scope:actor } }
                scope:actor = { add_courtier = prev }
            }
            # Transfer over any prisoners he was holding in the camp
            every_in_list = {
                list = strato_prisoners
                limit = { NOT = { liege_or_court_owner = scope:actor } }
                scope:actor = { imprison = prev }
            }
        }
        
        # Swear Fealty to the Byzantine Emperor
        create_title_and_vassal_change = {
            type = swear_fealty
            add_claim_on_loss = no
            save_scope_as = title_change
        }
        change_liege = {
            liege = scope:recipient
            change = scope:title_change
        }
        resolve_title_and_vassal_change = scope:title_change
        
        # Force Estate location and grant randomized starting buildings
        hidden_effect = {
            domicile ?= {
                move_domicile = title:c_byzantion.province
                
                # Baseline Estate Upgrades
                add_domicile_building = estate_main_01
                add_domicile_building = estate_main_02
                add_domicile_building = estate_main_03
                
                # Roll for Internal Specialized Rooms
                random_list = {
                    10 = {
                        add_domicile_building = servants_quarters_01
                        add_domicile_building = servants_quarters_02
                    }
                    10 = {
                        add_domicile_building = office_01
                        add_domicile_building = office_02
                    }
                    10 = {
                        add_domicile_building = prison_01
                        add_domicile_building = prison_02
                    }
                }
                
                # Roll for External Infrastructure
                random_list = {
                    10 = {
                        add_domicile_building = market_01
                        add_domicile_building = market_02
                    }
                    10 = {
                        add_domicile_building = workshop_01
                        add_domicile_building = workshop_02
                    }
                    10 = {
                        add_domicile_building = barracks_01
                        add_domicile_building = barracks_02
                    }
                }
            }
        }
    }
}

bookmark_ilione_nomadic_event_effect = {
    set_house = house:house_nars_1002

    add_character_flag = {
        flag = domicile_culture_and_faith_check
        days = 1
    }
    set_variable = {
        name = previous_provisions
        value = root.domicile.provisions
    }
    change_government = nomad_government

    hidden_effect = {
        create_title_and_vassal_change = {
            type = conquest
            save_scope_as = change
            add_claim_on_loss = no
        }
        title:c_tana = {
            change_title_holder = {
                holder = root
                change = scope:change
                take_baronies = yes
            }
        }
        title:c_kuzeytorg = {
            change_title_holder = {
                holder = root
                change = scope:change
                take_baronies = yes
            }
        }
        resolve_title_and_vassal_change = scope:change

        create_title_and_vassal_change = {
            type = granted
            save_scope_as = duchy_change
            add_claim_on_loss = no
        }
        title:d_tana = {
            change_title_holder = {
                holder = root
                change = scope:duchy_change
            }
        }
        resolve_title_and_vassal_change = scope:duchy_change
        
        # Herd mechanics and randomized starting yurt buildings
        domicile ?= {
            change_herd = {
                value = root.var:previous_provisions
                multiply = 0.25
            }
            
            # Baseline Yurt Upgrades[cite: 3]
            add_domicile_building = yurt_main_01
            add_domicile_building = yurt_main_02
            add_domicile_building = yurt_main_03
            add_domicile_building = yurt_main_04
            
            # Roll for External Warfare or Herd focus[cite: 3]
            random_list = {
                10 = {
                    add_domicile_building = mass_warfare_yurt_01
                    add_domicile_building = mass_warfare_yurt_02
                }
                10 = {
                    add_domicile_building = character_warfare_yurt_01
                    add_domicile_building = character_warfare_yurt_02
                }
                10 = {
                    add_domicile_building = herd_welfare_yurt_01
                    add_domicile_building = herd_welfare_yurt_02
                }
            }
            
            # Roll for Internal Main Yurt Upgrades[cite: 3]
            random_list = {
                10 = {
                    add_domicile_building = raid_main_yurt_01
                    add_domicile_building = raid_main_yurt_02
                }
                10 = {
                    add_domicile_building = herd_main_yurt_01
                    add_domicile_building = herd_main_yurt_02
                }
                10 = {
                    add_domicile_building = kurultai_main_yurt_01
                    add_domicile_building = kurultai_main_yurt_02
                }
            }
        }
    }

    set_primary_title_to = title:d_tana

    title:d_laamp_nars_1002 = {
        if = {
            limit = { exists = this }
            destroy_title = this
        }
    }
    
    custom_tooltip = adopt_nomadic_ways_decision_herd_tt
    if = {
        limit = { NOT = { has_trait = nomadic_philosophy } }
        add_trait = nomadic_philosophy
    }
}
