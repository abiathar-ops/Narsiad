local_assimilate_nikean_traditions_effect = {
	save_scope_as = converting_character
	
	$OLD_CULTURE$ = { save_scope_as = old_culture }
	$NEW_CULTURE$ = { save_scope_as = new_culture }
	$OLD_FAITH$ = { save_scope_as = old_faith }
	$NEW_FAITH$ = { save_scope_as = new_faith }
	$CONVERTER$ = { save_scope_as = converter }

	add_character_modifier = {
		modifier = embraced_nikean_traditions_modifier
	}
	
	if = {
		limit = {
			dynasty ?= {
				NOT = { has_dynasty_modifier = settled_nikean_dynasty_modifier }
			}
		}
		dynasty = {
			add_dynasty_modifier = { modifier = settled_nikean_dynasty_modifier years = 80 }
		}
	}
	
	set_culture = scope:new_culture
	if = {
		limit = { NOT = { faith = scope:new_faith } }
		hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
		set_character_faith = scope:new_faith
	}
	
	if = {
		limit = { is_ai = yes }
		add_character_flag = converted_culture_this_lifetime_ai
	}
	else = { add_character_flag = converted_culture_this_lifetime }

	every_close_family_member = {
		limit = { is_courtier_of = scope:converter }
		set_culture = scope:new_culture
		if = {
			limit = { NOT = { faith = scope:new_faith } }
			hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
			set_character_faith = scope:new_faith
		}
		hidden_effect = {
			add_character_modifier = { modifier = embraced_nikean_traditions_modifier }
		}
		add_to_temporary_list = notified_of_culture_change
	}
	
	every_consort = {
		limit = {
			is_ruler = no
			NOT = { is_in_list = notified_of_culture_change }
		}
		set_culture = scope:new_culture
		if = {
			limit = { NOT = { faith = scope:new_faith } }
			hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
			set_character_faith = scope:new_faith
		}
		hidden_effect = {
			add_character_modifier = { modifier = embraced_nikean_traditions_modifier }
		}
		add_to_temporary_list = notified_of_culture_change
	}

	every_vassal = {
		limit = { NOT = { is_in_list = notified_of_culture_change } }
		
		add_character_modifier = { modifier = embraced_nikean_traditions_modifier }
		
		hidden_effect = {
			if = {
				limit = {
					dynasty ?= {
						NOT = { has_dynasty_modifier = settled_nikean_dynasty_modifier }
					}
				}
				dynasty = {
					add_dynasty_modifier = { modifier = settled_nikean_dynasty_modifier years = 80 }
				}
			}
		}
		
		set_culture = scope:new_culture
		if = {
			limit = { NOT = { faith = scope:new_faith } }
			hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
			set_character_faith = scope:new_faith
		}

		hidden_effect = {
			every_close_family_member = {
				limit = { is_courtier_of = prev }
				set_culture = scope:new_culture
				if = {
					limit = { NOT = { faith = scope:new_faith } }
					hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
					set_character_faith = scope:new_faith
				}
				hidden_effect = { add_character_modifier = { modifier = embraced_nikean_traditions_modifier } }
				add_to_temporary_list = notified_of_culture_change
			}
			every_consort = {
				limit = {
					is_ruler = no
					NOT = { is_in_list = notified_of_culture_change }
				}
				set_culture = scope:new_culture
				if = {
					limit = { NOT = { faith = scope:new_faith } }
					hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
					set_character_faith = scope:new_faith
				}
				hidden_effect = { add_character_modifier = { modifier = embraced_nikean_traditions_modifier } }
				add_to_temporary_list = notified_of_culture_change
			}
		}

		add_to_temporary_list = notified_of_culture_change
	}

	hidden_effect = {
		every_player = {
			limit = {
				OR = {
					is_consort_of = scope:converting_character
					liege ?= scope:converting_character
					AND = {
						exists = scope:converting_character.liege
						this = scope:converting_character.liege
					}
				}
				NOT = { is_in_list = notified_of_culture_change }
			}
			send_interface_message = {
				type = msg_culture_conversion
				title = converted_culture_msg_title
				custom_tooltip = converted_from_old_culture_to_new_culture_tt
				right_icon = scope:converting_character
			}
		}
	}
}

create_h_narsokratia_effect = {
	save_scope_as = new_hegemon
	
	title:h_narsokratia = { save_scope_as = new_title }
	
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	hidden_effect = {
		scope:new_title = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
	}
	resolve_title_and_vassal_change = scope:change
	
	hidden_effect = {
		set_primary_title_to = scope:new_title
	}

	custom_tooltip = decision_h_narsokratia_troop_effect_tooltip
	
	hidden_effect = {
		spawn_army = {
			name = nikean_troop_hegemony_name
			men_at_arms = {
				type = maa_mastiges
				stacks = 15
			}
			men_at_arms = {
				type = maa_hoplitai
				stacks = 85
			}
			location = this.capital_province
			uses_supply = yes
			inheritable = yes
		}
	}

	scope:new_hegemon = {
		dynasty ?= { add_dynasty_prestige = 5000 }
		
		hidden_effect = {
			every_held_title = {
				limit = {
					tier = tier_kingdom
					NOT = { de_jure_liege = title:h_narsokratia }
				}
				set_de_jure_liege_title = title:h_narsokratia
			}
		}
	}
	
	hidden_effect = {
		every_kingdom = {
			limit = {
				is_titular = no
				NOT = { de_jure_liege = title:h_narsokratia }
				trigger_if = {
					limit = { exists = holder }
					holder != scope:new_hegemon
				}
				scope:new_hegemon = { completely_controls = prev }
			}
			set_de_jure_liege_title = title:h_narsokratia
		}
        nikean_hegemony_vassal_conversion_effect = yes
	}

	set_global_variable = {
		name = created_h_narsokratia
		value = yes
	}
}

pledge_restoration_effect = {
	if = {
		limit = {
			dynasty ?= {
				NOT = { has_dynasty_modifier = pledge_restoration_dynasty_modifier }
			}
		}
		dynasty = {
			add_dynasty_modifier = { modifier = pledge_restoration_dynasty_modifier }
		}
	}
	hidden_effect = { 
		dynasty = {
			every_dynasty_member = {
				limit = {
					is_alive = yes
					is_nikean = yes
					NOT = { has_trait = nars_scion }
				}
				add_trait = nars_scion
			}
		}
	}
}
