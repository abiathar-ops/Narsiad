local_assimilate_selian_traditions_effect = {
	save_scope_as = converting_character
	
	$OLD_CULTURE$ = { save_scope_as = old_culture }
	$NEW_CULTURE$ = { save_scope_as = new_culture }
	$OLD_FAITH$ = { save_scope_as = old_faith }
	$NEW_FAITH$ = { save_scope_as = new_faith }
	$CONVERTER$ = { save_scope_as = converter }

	add_character_modifier = {
		modifier = embraced_selian_traditions_modifier
	}
	
	if = {
		limit = {
			dynasty ?= {
				NOT = { has_dynasty_modifier = settled_selian_dynasty_modifier }
			}
		}
		dynasty = {
			add_dynasty_modifier = { modifier = settled_selian_dynasty_modifier years = 80 }
		}
	}
	
	set_culture = scope:new_culture
	if = {
		limit = { NOT = { faith = scope:new_faith } }
		hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
		set_character_faith = scope:new_faith
	}
	
	if = {
		limit = { is_ai = yes }
		add_character_flag = converted_culture_this_lifetime_ai
	}
	else = { add_character_flag = converted_culture_this_lifetime }

	every_close_family_member = {
		limit = { is_courtier_of = scope:converter }
		set_culture = scope:new_culture
		if = {
			limit = { NOT = { faith = scope:new_faith } }
			hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
			set_character_faith = scope:new_faith
		}
		hidden_effect = {
			add_character_modifier = { modifier = embraced_selian_traditions_modifier }
		}
		add_to_temporary_list = notified_of_culture_change
	}
	
	every_consort = {
		limit = {
			is_ruler = no
			NOT = { is_in_list = notified_of_culture_change }
		}
		set_culture = scope:new_culture
		if = {
			limit = { NOT = { faith = scope:new_faith } }
			hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
			set_character_faith = scope:new_faith
		}
		hidden_effect = {
			add_character_modifier = { modifier = embraced_selian_traditions_modifier }
		}
		add_to_temporary_list = notified_of_culture_change
	}

	every_vassal = {
		limit = { NOT = { is_in_list = notified_of_culture_change } }
		
		add_character_modifier = { modifier = embraced_selian_traditions_modifier }
		
		hidden_effect = {
			if = {
				limit = {
					dynasty ?= {
						NOT = { has_dynasty_modifier = settled_selian_dynasty_modifier }
					}
				}
				dynasty = {
					add_dynasty_modifier = { modifier = settled_selian_dynasty_modifier years = 80 }
				}
			}
		}
		
		set_culture = scope:new_culture
		if = {
			limit = { NOT = { faith = scope:new_faith } }
			hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
			set_character_faith = scope:new_faith
		}

		hidden_effect = {
			every_close_family_member = {
				limit = { is_courtier_of = prev }
				set_culture = scope:new_culture
				if = {
					limit = { NOT = { faith = scope:new_faith } }
					hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
					set_character_faith = scope:new_faith
				}
				hidden_effect = { add_character_modifier = { modifier = embraced_selian_traditions_modifier } }
				add_to_temporary_list = notified_of_culture_change
			}
			every_consort = {
				limit = {
					is_ruler = no
					NOT = { is_in_list = notified_of_culture_change }
				}
				set_culture = scope:new_culture
				if = {
					limit = { NOT = { faith = scope:new_faith } }
					hidden_effect = { bastard_to_wild_oat_conversion_assimilation_effect = yes }
					set_character_faith = scope:new_faith
				}
				hidden_effect = { add_character_modifier = { modifier = embraced_selian_traditions_modifier } }
				add_to_temporary_list = notified_of_culture_change
			}
		}

		add_to_temporary_list = notified_of_culture_change
	}

	hidden_effect = {
		every_player = {
			limit = {
				OR = {
					is_consort_of = scope:converting_character
					liege ?= scope:converting_character
					AND = {
						exists = scope:converting_character.liege
						this = scope:converting_character.liege
					}
				}
				NOT = { is_in_list = notified_of_culture_change }
			}
			send_interface_message = {
				type = msg_culture_conversion
				title = converted_culture_msg_title
				custom_tooltip = converted_from_old_culture_to_new_culture_tt
				right_icon = scope:converting_character
			}
		}
	}
}

create_h_narsokratia_effect = {
	# "The old one" = whatever the founder's own primary title was BEFORE this
	# effect runs (an empire, per decision_h_narsokratia's is_shown/is_valid).
	# Save it now, before set_primary_title_to overwrites it below, so the new
	# hegemony's name/adjective/color can be inherited from it.
	#
	# Note on government type: in this mod, government (oligarchic/aristocratic/
	# autocratic/etc, see common/governments/HAAO_government_types.txt) is a
	# CHARACTER attribute, not something a title can carry - h_narsokratia's old
	# static definition had no "government" field, and create_dynamic_title has
	# no such field either. Nothing in this effect ever calls
	# set_government_type, so root's government is never touched and simply
	# stays whatever it already was on the old realm - i.e. it "takes after the
	# old one" automatically, with nothing extra needed. (The only government
	# change anywhere in this decision is vassals being pushed to administrative
	# government by selian_vassal_conversion_effect at the very end - that's
	# unrelated to root and unaffected by this rewrite.)
	root = {
		primary_title = {
			save_scope_as = old_title
		}
	}

	save_scope_as = new_hegemon

	scope:new_hegemon = {
		give_nickname = nick_soter
		custom_tooltip = decision_h_narsokratia_effect_tooltip

		# Generate a brand-new hegemony-tier title instead of referencing the old
		# static h_narsokratia. name/adj point at loc keys (nars_titles_l_english.yml)
		# that read scope:old_title's own name/adjective, so the hegemony ends up
		# named after whatever realm the founder actually held, not a fixed label.
		# create_dynamic_title automatically populates scope:new_title, same as
		# vanilla's own hegemony-tier example in
		# common/scripted_effects/09_mpo_greatest_of_khans_effects.txt.
		hidden_effect = {
			create_dynamic_title = {
				tier = hegemony
				name = nars_hegemony_dynamic_name
				adj = nars_hegemony_dynamic_adj
				article = article_CONFEDERATION_CULTURE_NAME
			}
			scope:new_title = {
				set_color_from_title = scope:old_title
				set_definitive_form = yes
				set_can_be_named_after_dynasty = no
				set_can_use_nomadic_naming = no
				if = {
					limit = { exists = scope:new_hegemon.house }
					set_coa = scope:new_hegemon.house
				}
				else = {
					generate_coa = yes
				}
			}
		}

		custom_tooltip = decision_h_narsokratia_troop_effect_tooltip
		gain_heroic_legend_seed_tooltip_effect = yes
		dynasty ?= { add_dynasty_prestige = 5000 }
		add_prestige = 5000
		hidden_effect = {
			every_held_title = {
				limit = {
					tier = tier_kingdom
					NOT = { de_jure_liege = scope:new_title }
				}
				set_de_jure_liege_title = scope:new_title
			}
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		hidden_effect = {
			scope:new_title = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		
		hidden_effect = {
			set_primary_title_to = scope:new_title
		}
		
		hidden_effect = {
			spawn_army = {
				name = selian_troop_hegemony_name
				men_at_arms = {
					type = nars_t2
					stacks = 15
				}
				men_at_arms = {
					type = nars_t1
					stacks = 85
				}
				location = this.capital_province
				uses_supply = yes
				inheritable = yes
			}
		}
		
		hidden_effect = {
			every_kingdom = {
				limit = {
					is_titular = no
					NOT = { de_jure_liege = scope:new_title }
					trigger_if = {
						limit = { exists = holder }
						holder != scope:new_hegemon
					}
					scope:new_hegemon = { completely_controls = prev }
				}
				set_de_jure_liege_title = scope:new_title
			}
			selian_vassal_conversion_effect = yes
		}
		
	}
	
	set_global_variable = {
		name = created_h_narsokratia
		value = yes
	}
}
