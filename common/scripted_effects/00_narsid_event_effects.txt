# ─────────────────────────────────────────────────────────────────
#  TIERED VASSAL CONVERSION EFFECTS
#  Called by the three title-creation scripted effects.
#  Each tier converts more vassals and does more to their government.
# ─────────────────────────────────────────────────────────────────

# Kingdom tier: ~30 % chance per AI vassal ruler converts faith.
narsid_kingdom_vassal_conversion_effect = {
	every_vassal = {
		limit = { is_ai = yes }
		random = {
			chance = 30
			set_character_faith = root.faith
		}
	}
}

# Empire tier: ~60 % faith conversion + ALL vassals get administrative government.
narsid_empire_vassal_conversion_effect = {
	every_vassal = {
		# Faith: majority converts
		limit = { is_ai = yes }
		random = {
			chance = 60
			set_character_faith = root.faith
		}
	}
	# Government: all vassals flipped to administrative
	every_vassal = {
		limit = {
			NOT = { government_has_flag = government_is_administrative }
		}
		change_government = administrative_government
	}
}

# Hegemony tier: FULL government conversion + direct vassals' faith converted.
# Full culture/faith conversion of sub-vassals is handled by the
# separate decision_great_conversion decision.
narsid_hegemony_vassal_conversion_effect = {
	# Government: every vassal in the realm
	every_vassal = {
		limit = {
			NOT = { government_has_flag = government_is_administrative }
		}
		change_government = administrative_government
	}
	# Faith: direct vassals only (liege = root)
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		set_character_faith = root.faith
	}
}
