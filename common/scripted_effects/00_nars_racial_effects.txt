add_nikeis_trait_effect = {
    add_trait = nars_naturalis
}

apply_nikeis_trait = {
    if = {
        limit = {
            NOT = { has_character_flag = nars_naturalis_handled }
        }
        if = {
            limit = {
                age >= 0
                has_nikeis_blood = no
                is_ethnically_nikean = yes
            }
            add_nikeis_trait_effect = yes
        }
        else = {
            add_trait = nars_naturalis
        }
    }
}
