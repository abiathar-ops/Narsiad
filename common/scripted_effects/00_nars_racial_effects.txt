nars_naturalis_set_tier_xp_effect = {
    # 1. Wipe the track clean so values do not stack
    add_trait_xp = {
        trait = nars_naturalis
        track = polimetis
        value = -100
    }

    # 2. Apply the correct XP threshold based on Education Level
    if = {
        limit = { has_education_rank_5_trigger = yes } 
        add_trait_xp = {
            trait = nars_naturalis
            track = polimetis
            value = 100
        }
    }
    else_if = {
        limit = { has_education_rank_4_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polimetis
            value = 80
        }
    }
    else_if = {
        limit = { has_education_rank_3_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polimetis
            value = 60
        }
    }
    else_if = {
        limit = { has_education_rank_2_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polimetis
            value = 40
        }
    }
    else_if = {
        limit = { has_education_rank_1_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polimetis
            value = 20
        }
    }
}

nars_apply_setup_effect = {
    if = {
        limit = {
            OR = {
                is_ethnically_selian = yes
            }
            NOT = { has_trait = nars_naturalis }
        }
        add_trait = nars_naturalis
        #create_story = story_nars_naturalis
        nars_naturalis_set_tier_xp_effect = yes
    }
}
