nars_naturalis_set_tier_xp_effect = {
    # 1. Wipe the track clean so values do not stack
    add_trait_xp = {
        trait = nars_naturalis
        track = polymetis
        value = -100
    }

    # 2. Apply the correct XP threshold based on Education Level
    if = {
        limit = { has_education_rank_5_trigger = yes } 
        add_trait_xp = {
            trait = nars_naturalis
            track = polymetis
            value = 100
        }
    }
    else_if = {
        limit = { has_education_rank_4_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polymetis
            value = 80
        }
    }
    else_if = {
        limit = { has_education_rank_3_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polymetis
            value = 60
        }
    }
    else_if = {
        limit = { has_education_rank_2_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polymetis
            value = 40
        }
    }
    else_if = {
        limit = { has_education_rank_1_trigger = yes }
        add_trait_xp = {
            trait = nars_naturalis
            track = polymetis
            value = 20
        }
    }
}

nars_update_narsokratia_value = {
    # SCOPE: Story
    
    if = {
        # Catch Landless Adventurers first
        limit = { story_owner = { has_government = landless_adventurer_government } }
        set_variable = { name = narsokratia_value value = 0 }
    }
    else_if = {
        limit = { 
            story_owner = { 
                # Scoping to a title requires dropping the "tier_" prefix!
                any_held_title = { title_tier = hegemony } 
            } 
        }
        set_variable = { name = narsokratia_value value = 5 }
    }
    else_if = {
        limit = { 
            story_owner = { 
                any_held_title = { title_tier = empire } 
            } 
        }
        set_variable = { name = narsokratia_value value = 4 }
    }
    else_if = {
        limit = { 
            story_owner = { 
                any_held_title = { title_tier = kingdom } 
            } 
        }
        set_variable = { name = narsokratia_value value = 3 }
    }
    else_if = {
        limit = { 
            story_owner = { 
                any_held_title = { title_tier = duchy } 
            } 
        }
        set_variable = { name = narsokratia_value value = 2 }
    }
    else_if = {
        limit = { 
            story_owner = { 
                any_held_title = { title_tier = county } 
            } 
        }
        set_variable = { name = narsokratia_value value = 1 }
    }
    else = {
        # Catch-all for unlanded courtiers
        set_variable = { name = narsokratia_value value = 0 }
    }
}
