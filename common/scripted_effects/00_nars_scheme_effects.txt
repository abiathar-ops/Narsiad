nars_cap_schemes_and_fire_reminders_effect = {
	if = {
		limit = {
			scheme_agent_charges >= opportunity_cap_value
			scheme_success_chance >= max_scheme_success_chance
		}
		if = {
			limit = {
				scheme_owner = {
					is_ai = no
					NOR = {
						has_character_flag = scheme_critical_moments.0011
						has_variable = opportunities_reminder
					}
				}
			}
			scheme_owner = {
				add_character_flag = currently_in_scheme_reminder
				# supress the 'variable is used but is never set' error, the flag is set through code with the cooldown effect on the event, smh!
				if = {
					limit = {
						always = no
					}
					add_character_flag = scheme_critical_moments.0011
				}
			}
			switch = {
				trigger = scheme_type
				munitions_research_scheme = {
					munitions_research_scheme_prep_effect = yes
				}
				raise_warhost_scheme = {
					raise_warhost_scheme_prep_effect = yes
				}
			}
		}
		# If the player hasn't been reminded once, tell them that their scheme has hit max opportunities.
		scheme_owner = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_variable = opportunities_reminder }
				}
				trigger_event = scheme_critical_moments.0011
			}
		}
	}
}

munitions_research_scheme_prep_effect = {
	save_scope_as = scheme
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:nars_gunpowder.1011
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

munitions_research_success_effect = {
	scope:capital = {
		add_county_modifier = {
			modifier = munitions_research_modifier
			years = 30
		}
	}
	# Unlock Mástiges recruitment globally (checked in can_recruit_maa_02_trigger)
	set_global_variable = {
		name = mastiges_recruitable
		value = yes
	}
}

	}

		limit = {

		}

	}
}

create_narsil_effect = {
	if = {
		limit = {
			has_dlc_feature = royal_court
		}
		create_artifact = {
			name = narsil_name
			description = narsil_desc
			modifier = artifact_narsil_modifier
			template = narsil_template
			type = narsil
			visuals = narsil
            
			owner = root
			wealth = 3000 
			quality = 100 
			
			save_scope_as = newly_created_artifact
			history = {					
				type = created
			}
		}
        
        scope:newly_created_artifact = {
            set_artifact_rarity_illustrious = yes
			set_variable = { name = historical_unique_artifact value = yes }
			set_variable = {
				name = can_be_used_for_narsokratia
				value = yes
			}
        }
	}
}

get_item_quality_effect = {
	if = {
		limit = {
			NOT = { exists = scope:quality }
		}
		if = {
			limit = { NOT = { exists = scope:random_quality_bonus } }
			random_list = {
				10 = { # Illustrious — rarest
					modifier = {
						add = 15
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 100 }
						}
					}
					modifier = {
						add = 8
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 75 }
						}
					}
					save_scope_value_as = { name = random_quality_bonus value = 200 }
				}
				20 = { # Famed
					modifier = {
						add = 15
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 75 }
						}
					}
					modifier = {
						add = 8
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 50 }
						}
					}
					save_scope_value_as = { name = random_quality_bonus value = 80 }
				}
				30 = { # Masterwork
					modifier = {
						add = 10
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 50 }
						}
					}
					modifier = {
						add = 5
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 25 }
						}
					}
					save_scope_value_as = { name = random_quality_bonus value = 40 }
				}
				40 = { # Common — most likely
					modifier = {
						add = -20
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 75 }
						}
					}
					modifier = {
						add = -10
						scope:actor = {
							has_trait_track = { trait = lifestyle_pottery track = potter value >= 25 }
						}
					}
					save_scope_value_as = { name = random_quality_bonus value = 0 }
				}
			}
		}
		save_scope_value_as = {
			name = quality
			value = {
				value = 20
				add = scope:random_quality_bonus
				min = 1
				max = 200
			}
		}
	}
}

produce_amphora_effect = {
	$OWNER$ = { save_scope_as = owner }

	get_item_quality_effect = yes
	get_item_wealth_effect = yes

	hidden_effect_new_object = {
		create_artifact = {
			name = artifact_amphora_name
			description = artifact_placeholder_description
			type = amphora
			visuals = urn
			modifier = artifact_placeholder_modifier
			wealth = scope:wealth
			quality = scope:quality
			save_scope_as = newly_created_artifact
			history = {
				type = created
				recipient = scope:owner
				location = scope:owner.location
			}
		}
		scope:newly_created_artifact = {
			get_artifact_feature_references_effect = yes
			set_artifact_description = artifact_amphora_description
			add_scaled_artifact_modifier_minor_prestige_effect = yes
			if = {
				limit = {
					OR = {
						rarity = famed
						rarity = illustrious
					}
				}
				add_scaled_artifact_modifier_majesty_effect = yes
			}
			if = {
				limit = { rarity = illustrious }
				add_scaled_artifact_modifier_dynasty_prestige_effect = yes
			}
			set_variable = {
				name = quality
				value = scope:quality
			}
			set_variable = {
				name = wealth
				value = scope:wealth
			}
		}
	}
}
