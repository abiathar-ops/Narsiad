set_panteion_effect = {
	# Remove any previous personal god relation

	if = {
		limit = { is_orphic = yes }
		if = {	
			limit = {
				has_character_modifier = panteion_nikean_narses
			}
			add_piety = massive_piety_loss
			remove_character_modifier = panteion_nikean_narses
		}
		else_if = {	
			limit = {
				has_character_modifier = panteion_nikean_phokas
			}
			add_piety = massive_piety_loss
			remove_character_modifier = panteion_nikean_phokas
		}
		else_if = {	
			limit = {
				has_character_modifier = panteion_nikean_enuo
			}
			add_piety = massive_piety_loss
			remove_character_modifier = panteion_nikean_enuo
		}
		else = {
			add_piety = massive_piety_loss
		}
	}
	add_character_modifier = $PANTEION$
}

pledge_restoration_effect = {
	if = {
		limit = {
			dynasty ?= {
				NOT = { has_dynasty_modifier = pledge_restoration_dynasty_modifier }
			}
		}
		dynasty = {
			add_dynasty_modifier = { modifier = pledge_restoration_dynasty_modifier }
		}
	}
	custom_tooltip = nars_dynasty_get_scion_tt
	hidden_effect = { 
		dynasty = {
			every_dynasty_member = {
				limit = {
					is_alive = yes
					is_nikean = yes
					NOT = { has_trait = nars_scion }
				}
				add_trait = nars_scion
			}
		}
	}
}
