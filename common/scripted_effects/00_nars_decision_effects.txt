set_politeion_effect = {
	# Remove any previous personal god relation

	if = {
		limit = { is_archaic = yes }
		if = {	
			limit = {
				has_character_modifier = politeion_selian_narses
			}
			add_piety = massive_piety_loss
			remove_character_modifier = politeion_selian_narses
		}
		else_if = {	
			limit = {
				has_character_modifier = politeion_selian_phokas
			}
			add_piety = massive_piety_loss
			remove_character_modifier = politeion_selian_phokas
		}
		else_if = {	
			limit = {
				has_character_modifier = politeion_selian_enuo
			}
			add_piety = massive_piety_loss
			remove_character_modifier = politeion_selian_enuo
		}
		else = {
			add_piety = massive_piety_loss
		}
	}
	add_character_modifier = $POLITEION$
}
