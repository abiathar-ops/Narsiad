recruit_nars_scheme = {
	skill = diplomacy
	desc = "recruit_nars_scheme_desc_general"
	success_desc = "recruit_nars_scheme_SCHEME_SUCCESS_DESC"
	icon = icon_sacred
	illustration = "gfx/interface/illustrations/event_scenes/nars_temple.dds"
	category = personal
	target_type = character

	is_secret = no
	is_basic = yes

	hide_target_name = yes
	freeze_scheme_when_traveling = yes

	speed_per_skill_point = -2
	spymaster_speed_per_skill_point = 0
	uses_resistance = no
	base_progress_goal = 175
	base_maximum_success = 90
	minimum_success = 5

	allow = {
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = incapable }
		is_selian = yes
		is_landed_or_landless_administrative = yes
	}

	valid = {
		scope:target = scope:owner
		is_imprisoned = no
		is_at_war = no
		NOT = { has_trait = incapable }
		scope:owner = {
			is_selian = yes
			recruit_nars_court_has_room_trigger = yes
		}
	}

	odds_prediction = {
		add = odds_recruit_nars_scheme_misc_value
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 50
		# Skill
		modifier = {
			add = scope:owner.diplomacy
			desc = "SCHEME_MY_DIPLOMACY"
		}
		# Personality traits
		modifier = {
			add = 10
			desc = SCHEME_OWNER_GREGARIOUS
			scope:owner = { has_trait = gregarious }
		}
		modifier = {
			add = -10
			desc = SCHEME_OWNER_SHY
			scope:owner = { has_trait = shy }
		}
		modifier = {
			add = 10
			desc = SCHEME_OWNER_DILIGENT
			scope:owner = { has_trait = diligent }
		}
		modifier = {
			add = -10
			desc = SCHEME_OWNER_LAZY
			scope:owner = { has_trait = lazy }
		}
		modifier = {
			add = -10
			desc = SCHEME_OWNER_ARROGANT
			scope:owner = { has_trait = arrogant }
		}
		modifier = {
			add = 5
			desc = SCHEME_OWNER_HUMBLE
			scope:owner = { has_trait = humble }
		}
	}

	on_phase_completed = {
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }

		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}

		scope:owner = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = { on_action = recruit_nars_scheme_success }
			}
			else = {
				trigger_event = { on_action = recruit_nars_scheme_failure }
			}
		}
		reset_scheme_progress = yes
	}

	on_invalidated = {
	}
}
