piously_devote_scheme = {
	# Basic Setup
	skill = learning
	desc = "piously_devote_scheme_desc_general"
	success_desc = "piously_devote_scheme_SCHEME_SUCCESS_DESC"
	icon =	icon_sacred
	illustration = "gfx/interface/illustrations/event_scenes/nars_faith.dds"
	category = personal
	target_type = character

	is_secret = no
	is_basic = yes
	
	hide_target_name = yes
	freeze_scheme_when_traveling = yes

	# Parameters
	speed_per_skill_point = -2
	spymaster_speed_per_skill_point = 0	
	uses_resistance = no
	base_progress_goal = 225
	base_maximum_success = 95
	minimum_success = 5

	# Core Triggers
	allow = {
		age >= 12
		is_imprisoned = no
		NOT = { has_trait = incapable }
	}

	valid = {
		scope:target = scope:owner
		NOT = {
			has_character_modifier = partially_mute_modifier
		}
		is_imprisoned = no
		NOT = { has_trait = incapable }
	}

	odds_prediction = {
		add = odds_nars_scheme_misc_value
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 50
		# Skill
		modifier = {
			add = scope:owner.learning
			desc = "SCHEME_MY_LEARNING"
		}
		modifier = {
			add = -10
			desc = SCHEME_OWNER_LANDED
			scope:owner = { is_landed = yes }
		}
		modifier = {
			add = 10
			desc = SCHEME_OWNER_UNLANDED
			scope:owner = { is_landed = no }
		}
		# Personality traits
		modifier = {
			add = -10
			desc = SCHEME_OWNER_LAZY
			scope:owner = { has_trait = lazy }
		}
		modifier = {
			add = -30
			desc = SCHEME_OWNER_BURDENED
			scope:owner = { has_trait = burdened }
		}
		modifier = {
			add = -20
			desc = SCHEME_OWNER_CYNICAL
			scope:owner = { has_trait = cynical }
		}
		modifier = {
			add = 10
			desc = SCHEME_OWNER_DILIGENT
			scope:owner = { has_trait = diligent }
		}
		modifier = {
			add = -10
			desc = SCHEME_OWNER_IMPATIENT
			scope:owner = { has_trait = impatient }
		}
		modifier = {
			add = 10
			desc = SCHEME_OWNER_STUBBORN
			scope:owner = { has_trait = stubborn }
		}
		modifier = {
			add = 10
			desc = SCHEME_OWNER_ZEALOUS
			scope:owner = { has_trait = zealous }
		}
	}

	# On Actions
	on_phase_completed = {
		# Save scopes
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		# Let's see if we made progress
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		# HAAO - personal devotion level 3+ towards the tutelary deity: a bad show is half as likely. A failed roll gets
		# a second, even chance (scripted_triggers/HAAO_cultus_triggers.txt)
		if = {
			limit = {
				NOT = { exists = scope:scheme_successful }
				scope:owner = { deity_devotion_softens_bad_show = yes }
			}
			random = {
				chance = 50
				save_scope_value_as = {
					name = scheme_successful
					value = yes
				}
			}
		}
		# Determine the outcome
		scope:owner = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = { on_action = piously_devote_scheme_success }
			}
			else = {
				trigger_event = { on_action = piously_devote_scheme_failure }
			}
		}
		reset_scheme_progress = yes
	}

	## Invalidation
	on_invalidated = {
	}
}
