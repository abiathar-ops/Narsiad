produce_amphora_scheme = {
    skill = learning
    desc = "produce_amphora_scheme_desc_general"
    success_desc = "produce_amphora_scheme_SUCCESS_DESC"
    icon = icon_sacred #icon_devote_piously
    illustration = "gfx/interface/illustrations/event_scenes/nars_temple.dds" #haao_temple
    category = personal
    target_type = character

    is_secret = no
    is_basic = yes

    hide_target_name = yes

    speed_per_skill_point = -2
    spymaster_speed_per_skill_point = 0
    uses_resistance = no
    base_progress_goal = 175
    base_maximum_success = 90
    minimum_success = 5

    allow = {
        age >= 12
        is_imprisoned = no
        NOT = { has_trait = incapable }
    }

    valid = {
        scope:target = scope:owner
        is_imprisoned = no
        NOT = { has_trait = incapable }
        # NOT = {
        #     scope:owner = {
        #         var:produce_amphora_success >= 10
        #     }
        # }
    }

    odds_prediction = {
        add = odds_nars_scheme_misc_value
        min = 0
    }

    base_success_chance = {
        base = 30
        modifier = {
            add = scope:owner.prowess
            desc = "SCHEME_MY_PROWESS"
        }
        compare_modifier = {
            desc = "SCHEME_MY_LEARNING"
            value = scope:owner.learning
            multiplier = 0.25
        }
    }

    on_start = {
        set_variable = {
            name = apply_countermeasures
            value = flag:indirect
        }
    }

    on_phase_completed = {
        save_scope_as = scheme
        scheme_owner = {
            save_scope_as = owner
            random_owned_story = {
                limit = { story_type = story_produce_amphora }
                save_scope_as = amphora_story
            }
        }

        random = {
            chance = scope:scheme.scheme_success_chance
            save_scope_value_as = { name = scheme_successful value = yes }
        }

        scope:owner = {
            if = {
                limit = { exists = scope:scheme_successful }
                trigger_event = { on_action = produce_amphora_scheme_success }
                if = {
                    limit = { exists = scope:amphora_story }
                    scope:amphora_story = {
                        change_variable = {
                            name = produce_amphora_success
                            add = 1
                        }
                    }
                    scope:owner = {
                        set_variable = {
                            name = produce_amphora_success
                            value = scope:amphora_story.var:produce_amphora_success
                        }
                    }
                    if = {
                        limit = {
                            scope:owner = {
                                var:produce_amphora_success >= 10
                            }
                        }
                        scope:owner = {
                            trigger_event = nars_amphora.9000
                        }
                    }
                }
            }
            else = {
                trigger_event = { on_action = produce_amphora_scheme_failure }
            }
        }
    }

    on_invalidated = {
        scheme_owner = {
            save_scope_as = owner
            random_owned_story = {
                limit = { story_type = story_produce_amphora }
                save_scope_as = amphora_story
            }
        }
        if = {
            limit = { exists = scope:amphora_story }
            scope:amphora_story = {
                end_story = yes
            }
        }
    }
}
